1//! Skip Tag Registry
2//!
3//! Central registry for all skip tags with thread-safe access
4
5use super::{SkipTag, SkipTagId, TagRarity};
6use std::collections::HashMap;
7use std::sync::{Arc, RwLock};
8
9/// Thread-safe registry for all skip tags
10#[derive(Debug)]
11pub struct SkipTagRegistry {
12    tags: RwLock<HashMap<SkipTagId, Arc<dyn SkipTag>>>,
13}
14
15impl SkipTagRegistry {
16    /// Create a new empty registry
17    pub fn new() -> Self {
18        Self {
19            tags: RwLock::new(HashMap::new()),
20        }
21    }
22
23    /// Register a skip tag
24    pub fn register<T: SkipTag + 'static>(&self, tag: T) -> Result<(), String> {
25        let id = tag.id();
26        let tag_arc = Arc::new(tag);
27
28        let mut tags = self
29            .tags
30            .write()
31            .map_err(|_| "Failed to acquire write lock on skip tag registry")?;
32
33        if tags.contains_key(&id) {
34            return Err(format!("Skip tag {id} is already registered"));
35        }
36
37        tags.insert(id, tag_arc);
38        Ok(())
39    }
40
41    /// Get a skip tag by ID
42    pub fn get_tag(&self, id: SkipTagId) -> Option<Arc<dyn SkipTag>> {
43        let tags = self.tags.read().ok()?;
44        tags.get(&id).cloned()
45    }
46
47    /// Get all registered tag IDs
48    pub fn get_all_ids(&self) -> Vec<SkipTagId> {
49        let tags = self.tags.read().unwrap_or_else(|e| e.into_inner());
50        tags.keys().copied().collect()
51    }
52
53    /// Get tags by rarity
54    pub fn get_tags_by_rarity(&self, rarity: TagRarity) -> Vec<SkipTagId> {
55        let tags = self.tags.read().unwrap_or_else(|e| e.into_inner());
56        tags.values()
57            .filter(|tag| tag.rarity() == rarity)
58            .map(|tag| tag.id())
59            .collect()
60    }
61
62    /// Get weighted tag selection (for random generation)
63    pub fn get_weighted_tags(&self) -> Vec<(SkipTagId, f64)> {
64        let tags = self.tags.read().unwrap_or_else(|e| e.into_inner());
65        tags.values()
66            .map(|tag| (tag.id(), tag.rarity().weight()))
67            .collect()
68    }
69
70    /// Check if a tag is registered
71    pub fn is_registered(&self, id: SkipTagId) -> bool {
72        let tags = self.tags.read().unwrap_or_else(|e| e.into_inner());
73        tags.contains_key(&id)
74    }
75
76    /// Get the number of registered tags
77    pub fn count(&self) -> usize {
78        let tags = self.tags.read().unwrap_or_else(|e| e.into_inner());
79        tags.len()
80    }
81
82    /// Get all shop enhancement tags
83    pub fn get_all_shop_enhancement_tags(&self) -> Vec<SkipTagId> {
84        let tags = self.tags.read().unwrap_or_else(|e| e.into_inner());
85        tags.values()
86            .filter(|tag| tag.effect_type() == crate::skip_tags::TagEffectType::NextShopModifier)
87            .map(|tag| tag.id())
88            .collect()
89    }
90}
91
92impl Default for SkipTagRegistry {
93    fn default() -> Self {
94        Self::new()
95    }
96}

309    #[test]
310    fn test_weighted_tags() {
311        let registry = SkipTagRegistry::new();
312
313        registry
314            .register(MockTag {
315                id: SkipTagId::Double,
316                rarity: TagRarity::Common,
317            })
318            .unwrap();
319
320        registry
321            .register(MockTag {
322                id: SkipTagId::Boss,
323                rarity: TagRarity::Legendary,
324            })
325            .unwrap();
326
327        let weighted = registry.get_weighted_tags();
328        assert_eq!(weighted.len(), 2);
329
330        // Common should have higher weight than legendary
331        let double_weight = weighted
332            .iter()
333            .find(|(id, _)| *id == SkipTagId::Double)
334            .map(|(_, weight)| *weight)
335            .unwrap();
336        let boss_weight = weighted
337            .iter()
338            .find(|(id, _)| *id == SkipTagId::Boss)
339            .map(|(_, weight)| *weight)
340            .unwrap();
341
342        assert!(double_weight > boss_weight);
343    }
