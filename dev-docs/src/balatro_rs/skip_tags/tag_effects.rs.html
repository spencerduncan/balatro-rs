10/// Effect that gives immediate money reward
11pub fn money_effect(context: SkipTagContext, amount: i64) -> SkipTagResult {
12    let mut game = context.game;
13    game.money += amount as f64;
14
15    SkipTagResult {
16        game,
17        additional_tags: vec![],
18        success: true,
19        message: Some(format!("Gained ${amount}")),
20    }
21}

56/// Effect that duplicates another tag (Double tag)
57pub fn duplication_effect(context: SkipTagContext, selected_tag: SkipTagId) -> SkipTagResult {
58    // Exclude Double tags from duplication
59    if matches!(selected_tag, SkipTagId::Double) {
60        return SkipTagResult {
61            game: context.game,
62            additional_tags: vec![],
63            success: false,
64            message: Some("Cannot duplicate Double tags".to_string()),
65        };
66    }
67
68    SkipTagResult {
69        game: context.game,
70        additional_tags: vec![selected_tag],
71        success: true,
72        message: Some(format!("Duplicated {selected_tag} tag")),
73    }
74}

91/// Effect that upgrades a poker hand by levels
92pub fn hand_upgrade_effect(context: SkipTagContext, levels: u32) -> SkipTagResult {
93    let game = context.game;
94
95    // Get all available hand types that can be upgraded
96    let available_hands = get_upgradeable_hands(&game);
97
98    if available_hands.is_empty() {
99        return SkipTagResult {
100            game,
101            additional_tags: vec![],
102            success: false,
103            message: Some("No hands available to upgrade".to_string()),
104        };
105    }
106
107    // Select a random hand to upgrade
108    let random_index = game.rng.gen_range(0..available_hands.len());
109    let selected_hand = available_hands[random_index];
110
111    // TODO: Implement hand level upgrading
112    // This should upgrade the selected hand by the specified levels
113
114    SkipTagResult {
115        game,
116        additional_tags: vec![],
117        success: true,
118        message: Some(format!("Upgraded {selected_hand} by {levels} levels")),
119    }
120}

157#[cfg(test)]
158mod tests {
159    use super::*;
160    use crate::game::Game;
161    use crate::stage::Blind;
162
163    fn create_test_context() -> SkipTagContext {
164        SkipTagContext {
165            game: Game::default(),
166            skipped_blind: Some(Blind::Small),
167            available_tags: vec![SkipTagId::Boss, SkipTagId::Orbital],
168        }
169    }
170
171    #[test]
172    fn test_money_effect() {
173        let context = create_test_context();
174        let initial_money = context.game.money;
175
176        let result = money_effect(context, 50);
177
178        assert!(result.success);
179        assert_eq!(result.game.money, initial_money + 50.0);
180        assert!(result.message.unwrap().contains("$50"));
181    }
182
183    #[test]
184    fn test_duplication_effect_success() {
185        let context = create_test_context();
186
187        let result = duplication_effect(context, SkipTagId::Boss);
188
189        assert!(result.success);
190        assert_eq!(result.additional_tags.len(), 1);
191        assert_eq!(result.additional_tags[0], SkipTagId::Boss);
192    }
193
194    #[test]
195    fn test_duplication_effect_double_rejection() {
196        let context = create_test_context();
197
198        let result = duplication_effect(context, SkipTagId::Double);
199
200        assert!(!result.success);
201        assert!(result.additional_tags.is_empty());
202        assert!(result.message.unwrap().contains("Cannot duplicate Double"));
203    }
204
205    #[test]
206    fn test_hand_upgrade_effect() {
207        let context = create_test_context();
208
209        let result = hand_upgrade_effect(context, 3);
210
211        assert!(result.success);
212        let message = result.message.unwrap();
213        assert!(message.contains("Upgraded"));
214        assert!(message.contains("by 3 levels"));
215    }
216
217    #[test]
218    fn test_temporary_hand_size_effect() {
219        let context = create_test_context();
220
221        let result = temporary_hand_size_effect(context, 3);
222
