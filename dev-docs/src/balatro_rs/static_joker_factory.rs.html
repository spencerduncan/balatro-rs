12        Box::new(
13            StaticJoker::builder(JokerId::Joker, "Joker", "+4 Mult")
14                .rarity(JokerRarity::Common)
15                .cost(2)
16                .mult(4)
17                .per_hand()
18                .build()
19                .expect("Valid joker configuration"),
20        )
21    }
