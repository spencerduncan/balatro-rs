23/// Enhanced shop trait for generating shop contents with weighted randomization
24/// and support for various item types including jokers, consumables, vouchers, and packs.
25///
26/// This trait defines the core functionality for an advanced shop system that can:
27/// - Generate shops with multiple item types using weighted probabilities
28/// - Create booster packs with appropriate contents
29/// - Handle reroll mechanics with cost considerations
30/// - Adapt generation based on game state (ante, money, existing jokers, etc.)
31///
32/// # Design Philosophy
33///
34/// The shop generator follows a data-driven approach where game state influences
35/// the probability weights for different item types. This allows for:
36/// - Dynamic difficulty scaling based on ante progression
37/// - Adaptive item generation based on player resources
38/// - Extensible voucher effects that modify shop generation
39///
40/// # Implementation Example
41///
42/// ```rust,ignore
43/// use balatro_rs::shop::{ShopGenerator, EnhancedShop, ItemWeights};
44///
45/// struct StandardShopGenerator;
46///
47/// impl ShopGenerator for StandardShopGenerator {
48///     fn generate_shop(&self, game: &Game) -> EnhancedShop {
49///         let weights = self.calculate_weights(game);
50///         // Use weights to randomly select items for each slot
51///         EnhancedShop::new()
52///     }
53/// }
54/// ```
55pub trait ShopGenerator {
56    /// Generate a complete shop based on current game state.
57    ///
58    /// This method should create a new shop with items appropriate for the current
59    /// game state. Factors to consider:
60    /// - Current ante (affects item rarity and costs)
61    /// - Player money (affects affordability considerations)
62    /// - Existing jokers (avoid duplicates or create synergies)
63    /// - Active vouchers (that modify shop generation)
64    ///
65    /// # Arguments
66    /// * `game` - Current game state containing all relevant context
67    ///
68    /// # Returns
69    /// A fully populated `EnhancedShop` ready for player interaction
70    fn generate_shop(&self, game: &Game) -> EnhancedShop;
71
72    /// Generate a specific booster pack with appropriate contents.
73    ///
74    /// Pack contents should vary based on pack type and game state:
75    /// - Standard packs: Basic playing cards
76    /// - Jumbo/Mega packs: More cards with potential enhancements
77    /// - Spectral packs: Consumable cards with powerful effects
78    ///
79    /// # Arguments
80    /// * `pack_type` - The type of pack to generate
81    /// * `game` - Current game state for contextual generation
82    ///
83    /// # Returns
84    /// A `Pack` containing appropriate items for the specified type
85    fn generate_pack(&self, pack_type: PackType, game: &Game) -> Pack;
86
87    /// Calculate generation weights based on current game state.
88    ///
89    /// Weights determine the relative probability of each item type appearing
90    /// in shop generation. This method should consider:
91    /// - Game progression (higher ante = more rare items)
92    /// - Player resources (more money = higher cost items viable)
93    /// - Existing build synergies (recommend complementary items)
94    ///
95    /// # Arguments
96    /// * `game` - Current game state to analyze
97    ///
98    /// # Returns
99    /// `ItemWeights` struct containing probability weights for each item type
100    fn calculate_weights(&self, game: &Game) -> ItemWeights;
101
102    /// Reroll the shop contents while preserving shop state.
103    ///
104    /// This method should generate a new shop while maintaining:
105    /// - Reroll count and costs
106    /// - Any active shop modifiers
107    /// - Voucher effects on shop generation
108    ///
109    /// # Arguments
110    /// * `current_shop` - The shop being rerolled
111    /// * `game` - Current game state
112    ///
113    /// # Returns
114    /// A new `EnhancedShop` with fresh contents but preserved state
115    fn reroll_shop(&self, current_shop: &EnhancedShop, game: &Game) -> EnhancedShop;
116}

118/// Represents all possible items that can appear in the shop.
119///
120/// This enum encompasses every type of purchasable item in the enhanced shop system,
121/// providing a unified interface for shop generation, display, and purchase mechanics.
122///
123/// # Item Categories
124///
125/// - **Jokers**: Permanent modifiers that affect scoring and game mechanics
126/// - **Consumables**: Single-use cards with powerful immediate effects
127/// - **Vouchers**: Permanent upgrades that modify game rules or shop behavior
128/// - **Packs**: Containers with multiple items for batch purchasing
129/// - **Playing Cards**: Individual cards that can be added to the player's deck
130///
131/// # Design Considerations
132///
133/// Each item type has different:
134/// - **Cost structures**: Jokers use rarity-based pricing, packs have fixed costs
135/// - **Availability**: Some items may be limited by game progression
136/// - **Effects**: Items affect different aspects of the game (scoring, economy, deck composition)
137#[derive(Debug, Clone, PartialEq)]
138#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
139pub enum ShopItem {
140    /// A joker card identified by its ID.
141    ///
142    /// Jokers provide ongoing effects during gameplay and are typically
143    /// the most expensive and impactful shop items.
144    Joker(JokerId),
145
146    /// A consumable card (Tarot, Planet, Spectral).
147    ///
148    /// Consumables provide immediate effects when used and are consumed
149    /// in the process. They offer powerful but limited-use benefits.
150    Consumable(ConsumableType),
151
152    /// A specific consumable card identified by its ID.
153    ///
154    /// This provides access to individual consumable implementations rather
155    /// than generic placeholder types. Used by the tarot card system.
156    SpecificConsumable(ConsumableId),
157
158    /// A voucher that provides permanent upgrades.
159    ///
160    /// Vouchers modify game rules permanently, such as increasing shop
161    /// slots, reducing costs, or adding new mechanics.
162    Voucher(VoucherId),
163
164    /// A booster pack containing multiple cards.
165    ///
166    /// Packs provide value by containing multiple items at a potentially
167    /// discounted rate compared to individual purchases.
168    Pack(PackType),
169
170    /// A playing card that can be added to the deck.
171    ///
172    /// Individual playing cards allow precise deck customization and
173    /// can have special enhancements or properties.
174    PlayingCard(Card),
175}

177impl ShopItem {
178    /// Get the base cost of this item in coins.
179    ///
180    /// This provides the default cost before any modifiers are applied.
181    /// Actual shop prices may differ due to vouchers, sales, or other effects.
182    pub fn base_cost(&self, config: &crate::config::Config) -> usize {
183        match self {
184            ShopItem::Joker(joker_id) => {
185                // Would need to look up joker rarity, using placeholder for now
186                match joker_id {
187                    JokerId::Joker => 3, // Common joker base cost
188                    _ => 3,              // Default for now - would implement proper rarity lookup
189                }
190            }
191            ShopItem::Consumable(consumable_type) => match consumable_type {
192                ConsumableType::Tarot => 3,
193                ConsumableType::Planet => 3,
194                ConsumableType::Spectral => 4,
195            },
196            ShopItem::SpecificConsumable(consumable_id) => {
197                // Get base cost from tarot factory if it's a tarot card
198                use crate::consumables::tarot::get_tarot_factory;
199                if consumable_id.consumable_type() == crate::consumables::ConsumableType::Tarot {
200                    if let Ok(Some(metadata)) = get_tarot_factory().get_metadata(*consumable_id) {
201                        return metadata.rarity as usize + 2; // Common=2+2=4, Uncommon=3+2=5, etc.
202                    }
203                }
204                // Fallback to default consumable cost by type
205                match consumable_id.consumable_type() {
206                    crate::consumables::ConsumableType::Tarot => 3,
207                    crate::consumables::ConsumableType::Planet => 3,
208                    crate::consumables::ConsumableType::Spectral => 4,
209                }
210            }
211            ShopItem::Voucher(_) => 10, // Standard voucher cost
212            ShopItem::Pack(pack_type) => pack_type.base_cost(config),
213            ShopItem::PlayingCard(_) => 2, // Standard playing card cost
214        }
215    }
216
217    /// Get a human-readable name for this item.
218    pub fn display_name(&self) -> String {
219        match self {
220            ShopItem::Joker(joker_id) => format!("{joker_id:?} Joker"),
221            ShopItem::Consumable(consumable_type) => format!("{consumable_type:?} Card"),
222            ShopItem::SpecificConsumable(consumable_id) => format!("{consumable_id}"),
223            ShopItem::Voucher(voucher_id) => format!("{voucher_id:?} Voucher"),
224            ShopItem::Pack(pack_type) => format!("{pack_type:?} Pack"),
225            ShopItem::PlayingCard(card) => format!("{card}"),
226        }
227    }
228
229    /// Check if this item type is affected by specific voucher effects.
230    pub fn is_affected_by_voucher(&self, voucher: VoucherId) -> bool {
231        match voucher {
232            VoucherId::Overstock => true,     // Affects all shop items
233            VoucherId::ClearanceSale => true, // 50% off all items in shop
234            VoucherId::Liquidation => true,   // 25% off all items in shop
235            _ => false,
236        }
237    }
238}

314/// Enhanced shop with support for multiple item types and weighted generation
315#[derive(Debug, Clone)]
316pub struct EnhancedShop {
317    /// All slots in the shop
318    pub slots: Vec<ShopSlot>,
319    /// Number of rerolls available
320    pub rerolls_remaining: usize,
321    /// Cost per reroll
322    pub reroll_cost: usize,
323    /// Generation weights used for this shop
324    pub weights: ItemWeights,
325}
326
327impl EnhancedShop {
328    /// Create a new empty enhanced shop
329    pub fn new() -> Self {
330        Self {
331            slots: Vec::new(),
332            rerolls_remaining: 3,
333            reroll_cost: 5,
334            weights: ItemWeights::default(),
335        }
336    }
337
338    /// Check if a specific item is available in the shop
339    pub fn has_item(&self, item: &ShopItem) -> bool {
340        self.slots
341            .iter()
342            .any(|slot| slot.available && slot.item == *item)
343    }
344
345    /// Get the cost of a specific item if available
346    pub fn get_item_cost(&self, item: &ShopItem) -> Option<usize> {
347        self.slots
348            .iter()
349            .find(|slot| slot.available && slot.item == *item)
350            .map(|slot| slot.cost)
351    }
352
353    /// Purchase an item from the shop
354    pub fn purchase_item(&mut self, item: &ShopItem) -> Result<ShopItem, GameError> {
355        let slot_index = self
356            .slots
357            .iter()
358            .position(|slot| slot.available && slot.item == *item)
359            .ok_or(GameError::InvalidAction)?;
360
361        self.slots[slot_index].available = false;
362        Ok(item.clone())
363    }
364}
365
366impl Default for EnhancedShop {
367    fn default() -> Self {
368        Self::new()
369    }
370}
