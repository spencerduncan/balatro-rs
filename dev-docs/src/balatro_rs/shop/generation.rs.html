43impl RerollMechanics for StandardRerollMechanics {
44    /// Calculate the cost of a reroll based on current cost and game state
45    ///
46    /// Reroll costs follow this escalation pattern:
47    /// - Base cost: 5 coins
48    /// - First reroll: 5 coins
49    /// - Second reroll: 10 coins (+5)
50    /// - Third reroll: 15 coins (+5)
51    /// - Each subsequent reroll: +5 coins
52    ///
53    /// This provides a reasonable escalation that discourages excessive rerolling
54    /// while still allowing strategic use of the mechanic.
55    fn calculate_reroll_cost(&self, current_cost: usize, game: &Game) -> usize {
56        // Base reroll cost escalation: each reroll increases cost by 5
57        let base_increase = 5;
58        let new_cost = current_cost + base_increase;
59
60        // Apply voucher effects after base calculation
61        let shop_vouchers: Vec<VoucherId> = self.get_active_shop_vouchers(game);
62        self.apply_voucher_effects(new_cost, &shop_vouchers)
63    }
64
65    /// Check if a reroll is available based on shop state and game resources
66    fn can_reroll(&self, shop: &EnhancedShop, game: &Game) -> bool {
67        // Check if rerolls are remaining (if limited)
68        if shop.rerolls_remaining == 0 {
69            return false;
70        }
71
72        // Check if player has enough money for the reroll
73        if (shop.reroll_cost as f64) > game.money {
74            return false;
75        }
76
77        true
78    }
79
80    /// Apply voucher effects to reroll cost
81    fn apply_voucher_effects(&self, base_cost: usize, vouchers: &[VoucherId]) -> usize {
82        let mut final_cost = base_cost as f64;
83
84        for &voucher in vouchers {
85            match voucher {
86                VoucherId::Reroll => {
87                    // Reroll voucher provides free rerolls or reduces cost
88                    final_cost = 0.0; // Makes rerolls free
89                }
90                VoucherId::Liquidation => {
91                    // Liquidation gives 20% discount on all shop operations
92                    final_cost *= 0.8;
93                }
94                _ => {} // Other vouchers don't affect reroll costs
95            }
96        }
97
98        // Ensure minimum cost of 0 (free rerolls are possible)
99        (final_cost.round() as usize).max(0)
100    }
101}

156    /// Get joker rarity weights based on ante progression
157    #[allow(dead_code)]
158    fn get_joker_rarity_weights(&self, ante: usize) -> [f64; 4] {
159        // Base weights from issue requirements
160        let base_common = 70.0;
161        let base_uncommon = 25.0;
162        let base_rare = 4.5;
163        let base_legendary = 0.5;
164
165        // Ante scaling: higher antes increase rare chances
166        let scaling_factor = 1.0 + (ante as f64 - 1.0) * 0.1;
167
168        let scaled_rare = base_rare * scaling_factor;
169        let scaled_legendary = base_legendary * scaling_factor;
170
171        // Reduce common/uncommon proportionally
172        let total_increased = (scaled_rare - base_rare) + (scaled_legendary - base_legendary);
173        let reduction_factor = total_increased / (base_common + base_uncommon);
174
175        let scaled_common = (base_common * (1.0 - reduction_factor)).max(40.0);
176        let scaled_uncommon = (base_uncommon * (1.0 - reduction_factor)).max(15.0);
177
178        [
179            scaled_common,
180            scaled_uncommon,
181            scaled_rare,
182            scaled_legendary,
183        ]
184    }

217    /// Generate a weighted random joker based on rarity distribution
218    #[allow(dead_code)]
219    fn generate_weighted_joker(&mut self, ante: usize) -> Option<JokerId> {
220        let weights = self.get_joker_rarity_weights(ante);
221        let rarities = [
222            JokerRarity::Common,
223            JokerRarity::Uncommon,
224            JokerRarity::Rare,
225            JokerRarity::Legendary,
226        ];
227
228        // Select rarity using weighted distribution
229        let selected_rarity = self
230            .rng
231            .choose_weighted(&rarities, |r| match r {
232                JokerRarity::Common => weights[0],
233                JokerRarity::Uncommon => weights[1],
234                JokerRarity::Rare => weights[2],
235                JokerRarity::Legendary => weights[3],
236            })
237            .unwrap_or(&JokerRarity::Common);
238
239        // Get available jokers for this rarity
240        let available_jokers = self.get_jokers_by_rarity(*selected_rarity);
241        if available_jokers.is_empty() {
242            return None;
243        }
244
245        // Randomly select from available jokers of this rarity
246        let joker_index = self.rng.gen_range(0..available_jokers.len());
247        Some(available_jokers[joker_index])
248    }

380    fn generate_shop(&self, game: &Game) -> EnhancedShop {
381        let mut shop = EnhancedShop::new();
382        let weights = self.calculate_weights(game);
383        shop.weights = weights.clone();
384
385        // Standard shop has 5 slots
386        const SHOP_SLOTS: usize = 5;
387
388        // Create weighted distribution for item types
389        let item_weights = [
390            weights.joker_weight,
391            weights.consumable_weight,
392            weights.voucher_weight,
393            weights.pack_weight,
394            weights.playing_card_weight,
395        ];
396
397        for _ in 0..SHOP_SLOTS {
398            let item_type_index = self
399                .rng
400                .choose_weighted(&[0, 1, 2, 3, 4], |&i| item_weights[i])
401                .unwrap_or(&0);
402
403            let item = match item_type_index {
404                0 => self.generate_random_joker(game),
405                1 => self.generate_random_consumable(),
406                2 => self.generate_random_voucher(),
407                3 => self.generate_random_pack(),
408                4 => self.generate_random_playing_card(),
409                _ => self.generate_random_joker(game), // Fallback
410            };
411
412            if let Some(shop_item) = item {
413                let base_cost = shop_item.base_cost(&game.config);
414                // For now, use empty voucher list since voucher system is placeholder
415                let shop_vouchers: Vec<VoucherId> = vec![];
416                let final_cost = self.calculate_final_cost(base_cost, &shop_vouchers);
417
418                shop.slots.push(ShopSlot {
419                    item: shop_item,
420                    cost: final_cost,
421                    available: true,
422                    modifiers: vec![],
423                });
424            }
425        }
426
427        shop

542    fn calculate_weights(&self, game: &Game) -> ItemWeights {
543        let ante_number = self.ante_to_number(game.ante_current);
544        let money = game.money;
545
546        // For now, use empty voucher list since voucher system is placeholder
547        // In future, would convert from game vouchers to shop vouchers
548        let shop_vouchers: Vec<VoucherId> = vec![];
549
550        // Create cache key for performance optimization (currently unused)
551        let _cache_key = CacheKey {
552            ante: ante_number,
553            money,
554            vouchers: shop_vouchers.clone(),
555        };
556
557        // Check cache first (in real implementation, would need mutable access to use cache)
558        // For now, always calculate fresh weights
559
560        // Start with default weights
561        let mut weights = ItemWeights::default();
562
563        // Apply ante-based scaling
564        // Higher antes should favor higher-value items
565        let ante_scale = 1.0 + (ante_number as f64 * 0.15);
566
567        // Adjust weights based on ante progression
568        if ante_number >= 3 {
569            weights.voucher_weight *= ante_scale * 0.8; // Vouchers become more valuable
570            weights.pack_weight *= ante_scale * 0.9; // Packs become more valuable
571        }
572
573        if ante_number >= 5 {
574            weights.joker_weight *= ante_scale; // Jokers become even more important
575            weights.playing_card_weight *= 0.7; // Playing cards less important in late game
576        }
577
578        // Apply money-based adjustments
579        // If player has lots of money, bias toward expensive items
580        if money >= 50.0 {
581            weights.voucher_weight *= 1.3;
582            weights.pack_weight *= 1.2;
583        } else if money <= 20.0 {
584            // If low on money, favor cheaper items
585            weights.playing_card_weight *= 1.4;
586            weights.consumable_weight *= 1.2;
587        }
588
589        // Apply voucher effects
590        weights = self.apply_voucher_effects(weights, &shop_vouchers);
591
592        weights
593    }

595    fn reroll_shop(&self, current_shop: &EnhancedShop, game: &Game) -> EnhancedShop {
596        // Check if reroll is possible
597        if !self.reroll_mechanics.can_reroll(current_shop, game) {
598            // Return current shop unchanged if reroll not possible
599            return current_shop.clone();
600        }
601
602        // Generate new shop contents while preserving shop state
603        let mut new_shop = self.generate_shop(game);
604
605        // Calculate new reroll cost using the mechanics
606        let new_reroll_cost = self
607            .reroll_mechanics
608            .calculate_reroll_cost(current_shop.reroll_cost, game);
609
610        // Preserve and update shop state from the current shop
611        new_shop.reroll_cost = new_reroll_cost;
612
613        // Handle reroll count - for now we'll use unlimited rerolls (common in Balatro)
614        // In the future, this could be limited by vouchers or game settings
615        new_shop.rerolls_remaining = current_shop.rerolls_remaining.saturating_sub(0); // Unlimited for now
616
617        // Preserve generation weights from current shop for consistency
618        new_shop.weights = current_shop.weights.clone();
619
620        new_shop
621    }

820    #[test]
821    fn test_statistical_distribution_common_jokers() {
822        let mut generator = WeightedGenerator::new();
823        let mut rarity_counts = HashMap::new();
824        let sample_size = 1000;
825
826        // Get actual joker lists from factory for mapping
827        let common_jokers = JokerFactory::get_by_rarity(JokerRarity::Common);
828        let uncommon_jokers = JokerFactory::get_by_rarity(JokerRarity::Uncommon);
829        let rare_jokers = JokerFactory::get_by_rarity(JokerRarity::Rare);
830        let legendary_jokers = JokerFactory::get_by_rarity(JokerRarity::Legendary);
831
832        // Generate large sample for statistical testing
833        for _ in 0..sample_size {
834            if let Some(joker) = generator.generate_weighted_joker(1) {
835                let rarity = if common_jokers.contains(&joker) {
836                    JokerRarity::Common
837                } else if uncommon_jokers.contains(&joker) {
838                    JokerRarity::Uncommon
839                } else if rare_jokers.contains(&joker) {
840                    JokerRarity::Rare
841                } else if legendary_jokers.contains(&joker) {
842                    JokerRarity::Legendary
843                } else {
844                    JokerRarity::Common // Fallback
845                };
846                *rarity_counts.entry(rarity).or_insert(0) += 1;
847            }
848        }
849
850        let total_generated = rarity_counts.values().sum::<usize>() as f64;
851
852        // Check if distribution is approximately correct (within 10% tolerance)
853        if let Some(&common_count) = rarity_counts.get(&JokerRarity::Common) {
854            let common_percentage = (common_count as f64 / total_generated) * 100.0;
855            assert!(common_percentage > 60.0 && common_percentage < 80.0); // 70% ± 10%
856        }
857
858        if let Some(&uncommon_count) = rarity_counts.get(&JokerRarity::Uncommon) {
859            let uncommon_percentage = (uncommon_count as f64 / total_generated) * 100.0;
860            assert!(uncommon_percentage > 15.0 && uncommon_percentage < 35.0); // 25% ± 10%
861        }
862    }
