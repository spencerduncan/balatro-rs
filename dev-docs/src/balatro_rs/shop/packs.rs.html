157    /// Generate pack contents based on pack type, game state, and configuration
158    pub fn generate_contents(&mut self, game: &Game, config: &Config) -> Result<(), GameError> {
159        let (min_options, max_options) = self.pack_type.option_count(config);
160        let option_count = if min_options == max_options {
161            min_options
162        } else {
163            // Use proper randomization for variable option counts
164            game.rng.gen_range(min_options..=max_options)
165        };
166
167        // Pack option modifiers would be checked here in future implementations
168
169        self.options.clear();
170
171        match self.pack_type {
172            PackType::Standard | PackType::Jumbo | PackType::Mega | PackType::Enhanced => {
173                self.generate_standard_options(option_count, game, config)?
174            }
175            PackType::Variety => {
176                // Variety packs have mixed contents - for now just use standard
177                self.generate_standard_options(option_count, game, config)?
178            }
179            PackType::Buffoon | PackType::MegaBuffoon => {
180                self.generate_buffoon_options(option_count, game, config)?
181            }
182            PackType::Arcana | PackType::MegaArcana => {
183                self.generate_arcana_options(option_count, game, config)?
184            }
185            PackType::Celestial | PackType::MegaCelestial => {
186                self.generate_celestial_options(option_count, game, config)?
187            }
188            PackType::Spectral | PackType::MegaSpectral => {
189                self.generate_spectral_options(option_count, game, config)?
190            }
191        }
192
193        Ok(())
194    }

196    /// Generate standard pack options (playing cards)
197    fn generate_standard_options(
198        &mut self,
199        count: usize,
200        game: &Game,
201        config: &Config,
202    ) -> Result<(), GameError> {
203        use crate::card::{Enhancement, Suit, Value};
204
205        let suits = [Suit::Heart, Suit::Diamond, Suit::Club, Suit::Spade];
206        let values = [
207            Value::Ace,
208            Value::Two,
209            Value::Three,
210            Value::Four,
211            Value::Five,
212            Value::Six,
213            Value::Seven,
214            Value::Eight,
215            Value::Nine,
216            Value::Ten,
217            Value::Jack,
218            Value::Queen,
219            Value::King,
220        ];
221
222        for _ in 0..count {
223            // Select random suit and value
224            let suit = *game.rng.choose(&suits).unwrap();
225            let value = *game.rng.choose(&values).unwrap();
226
227            // Create card with potential enhancement (configurable chance)
228            let mut card = Card::new(value, suit);
229            if game.rng.gen_bool(config.enhancement_rate) {
230                let enhancements = [
231                    Enhancement::Bonus,
232                    Enhancement::Mult,
233                    Enhancement::Wild,
234                    Enhancement::Glass,
235                    Enhancement::Steel,
236                ];
237                card.enhancement = Some(*game.rng.choose(&enhancements).unwrap());
238            }
239
240            let enhancement_prefix = match card.enhancement {
241                Some(enh) => format!("{enh:?} "),
242                None => String::new(),
243            };
244
245            let option = PackOption::new(
246                ShopItem::PlayingCard(card),
247                format!("{enhancement_prefix}{value:?} of {suit:?}"),
248            );
249            self.options.push(option);
250        }
251
252        Ok(())
253    }

255    /// Generate buffoon pack options (jokers)
256    fn generate_buffoon_options(
257        &mut self,
258        count: usize,
259        game: &Game,
260        config: &Config,
261    ) -> Result<(), GameError> {
262        use crate::joker::JokerRarity;
263
264        // Define rarity weights using configuration values
265        let rarities = [
266            (JokerRarity::Common, config.joker_rarity_weight_common),
267            (JokerRarity::Uncommon, config.joker_rarity_weight_uncommon),
268            (JokerRarity::Rare, config.joker_rarity_weight_rare),
269        ];
270
271        // Available jokers by rarity (using the jokers we know exist)
272        let common_jokers = [
273            JokerId::Joker,
274            JokerId::GreedyJoker,
275            JokerId::LustyJoker,
276            JokerId::WrathfulJoker,
277            JokerId::GluttonousJoker,
278            JokerId::JollyJoker,
279        ];
280
281        let uncommon_jokers = [
282            JokerId::ZanyJoker,
283            JokerId::MadJoker,
284            JokerId::CrazyJoker,
285            JokerId::DrollJoker,
286        ];
287
288        let rare_jokers = [
289            JokerId::SlyJoker,
290            JokerId::WilyJoker,
291            JokerId::CleverJoker,
292            JokerId::DeviousJoker,
293        ];
294
295        for _ in 0..count {
296            // Select rarity based on weighted distribution
297            let total_weight: u32 = rarities.iter().map(|(_, weight)| weight).sum();
298            let roll = game.rng.gen_range(1..=total_weight);
299
300            let mut current_weight = 0;
301            let selected_rarity = rarities
302                .iter()
303                .find(|(_, weight)| {
304                    current_weight += weight;
305                    roll <= current_weight
306                })
307                .map(|(rarity, _)| *rarity)
308                .unwrap_or(JokerRarity::Common);
309
310            // Select joker based on rarity
311            let joker_id = match selected_rarity {
312                JokerRarity::Common => *game.rng.choose(&common_jokers).unwrap(),
313                JokerRarity::Uncommon => *game.rng.choose(&uncommon_jokers).unwrap(),
314                JokerRarity::Rare => *game.rng.choose(&rare_jokers).unwrap(),
315                JokerRarity::Legendary => JokerId::Joker, // Fallback to basic for legendary
316            };
317
318            let option = PackOption::new(
319                ShopItem::Joker(joker_id),
320                format!("{joker_id:?} Joker ({selected_rarity:?})"),
321            );
322            self.options.push(option);
323        }
324
325        Ok(())
326    }

411    /// Select an option from this pack
412    pub fn select_option(&self, option_index: usize) -> Result<ShopItem, GameError> {
413        if !self.is_valid_selection(option_index) {

438/// Trait for pack generation and management
439pub trait PackGenerator {
440    /// Generate a pack of the specified type
441    fn generate_pack(&self, pack_type: PackType, game: &Game) -> Result<Pack, GameError>;
442
443    /// Check if a pack type is available for purchase
444    fn is_pack_available(&self, pack_type: PackType, game: &Game) -> bool;
445
446    /// Get all available pack types for current game state
447    fn available_pack_types(&self, game: &Game) -> Vec<PackType>;
448}
449
450/// Default pack generator implementation
451pub struct DefaultPackGenerator;
452
453impl PackGenerator for DefaultPackGenerator {
454    fn generate_pack(&self, pack_type: PackType, game: &Game) -> Result<Pack, GameError> {
455        let config = &game.config;
456        let mut pack = Pack::new(pack_type, config);
457        pack.generate_contents(game, config)?;
458        Ok(pack)
459    }
460
461    fn is_pack_available(&self, pack_type: PackType, game: &Game) -> bool {
462        // Basic availability logic - all packs available if player has money
463        game.money >= pack_type.base_cost(&game.config) as f64
464    }
465
466    fn available_pack_types(&self, game: &Game) -> Vec<PackType> {
467        let all_pack_types = [
468            PackType::Standard,
469            PackType::Buffoon,
470            PackType::Arcana,
471            PackType::Celestial,
472            PackType::Spectral,
473            PackType::MegaBuffoon,
474            PackType::MegaArcana,
475            PackType::MegaCelestial,
476            PackType::MegaSpectral,
477        ];
478
479        all_pack_types
480            .iter()
481            .filter(|&&pack_type| self.is_pack_available(pack_type, game))
482            .copied()
483            .collect()
484    }
