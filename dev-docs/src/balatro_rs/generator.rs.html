9    // Get all legal SelectCard actions that can be executed given current state
10    fn gen_actions_select_card(&self) -> Option<impl Iterator<Item = Action>> {
11        // Can only select card during blinds
12        if !self.stage.is_blind() {
13            return None;
14        }
15        // Cannot select more than max
16        if self.available.selected().len() >= self.config.selected_max {
17            return None;
18        }
19        let combos = self
20            .available
21            .not_selected()
22            .clone()
23            .into_iter()
24            .map(Action::SelectCard);
25        Some(combos)
26    }

28    // Get all legal Play actions that can be executed given current state
29    fn gen_actions_play(&self) -> Option<impl Iterator<Item = Action>> {
30        // Can only play hand during blinds
31        if !self.stage.is_blind() {
32            return None;
33        }
34        // If no plays remaining, return None
35        if self.plays == 0.0 {
36            return None;
37        }
38        // If no cards selected, return None
39        if self.available.selected().is_empty() {
40            return None;
41        }
42        let combos = vec![Action::Play()].into_iter();
43        Some(combos)
44    }

46    // Get all legal Play actions that can be executed given current state
47    fn gen_actions_discard(&self) -> Option<impl Iterator<Item = Action>> {
48        // Can only discard during blinds
49        if !self.stage.is_blind() {
50            return None;
51        }
52        // If no discards remaining, return None
53        if self.discards == 0.0 {
54            return None;
55        }
56        // If no cards selected, return None
57        if self.available.selected().is_empty() {
58            return None;
59        }
60        let combos = vec![Action::Discard()].into_iter();
61        Some(combos)
62    }

64    // Get all legal move card actions
65    fn gen_actions_move_card(&self) -> Option<impl Iterator<Item = Action>> {
66        // Can only move cards during blinds
67        if !self.stage.is_blind() {
68            return None;
69        }
70        let left = self
71            .available
72            .cards()
73            .clone()
74            .into_iter()
75            .skip(1)
76            .map(|c| Action::MoveCard(MoveDirection::Left, c));
77        let right = self
78            .available
79            .cards()
80            .clone()
81            .into_iter()
82            .rev()
83            .skip(1)
84            .rev()
85            .map(|c| Action::MoveCard(MoveDirection::Right, c));
86
87        let combos = left.chain(right);
88        Some(combos)
89    }

91    // Get cash out action
92    fn gen_actions_cash_out(&self) -> Option<impl Iterator<Item = Action>> {
93        // If stage is not post blind, cannot cash out
94        if self.stage != Stage::PostBlind() {
95            return None;
96        }
97        Some(vec![Action::CashOut(self.reward)].into_iter())
98    }

100    // Get next round action
101    fn gen_actions_next_round(&self) -> Option<impl Iterator<Item = Action>> {
102        // If stage is not shop, cannot next round
103        if self.stage != Stage::Shop() {
104            return None;
105        }
106        Some(vec![Action::NextRound()].into_iter())
107    }

109    // Get select blind action
110    fn gen_actions_select_blind(&self) -> Option<impl Iterator<Item = Action>> {
111        // If stage is not pre blind, cannot select blind
112        if self.stage != Stage::PreBlind() {
113            return None;
114        }
115        if let Some(blind) = self.blind {
116            Some(vec![Action::SelectBlind(blind.next())].into_iter())
117        } else {
118            Some(vec![Action::SelectBlind(Blind::Small)].into_iter())
119        }
120    }

122    // Get buy joker actions
123    fn gen_actions_buy_joker(&self) -> Option<impl Iterator<Item = Action> + use<'_>> {
124        // If stage is not shop, cannot buy
125        if self.stage != Stage::Shop() {
126            return None;
127        }
128        // Cannot buy if all joker slots full
129        if self.joker_count() >= self.config.joker_slots {
130            return None;
131        }
132        self.shop
133            .gen_moves_buy_joker(self.money, self.jokers.len(), self.config.joker_slots)
134    }

136    // Get buy pack actions
137    fn gen_actions_buy_pack(&self) -> Option<impl Iterator<Item = Action> + use<'_>> {
138        // If stage is not shop, cannot buy packs
139        if self.stage != Stage::Shop() {
140            return None;
141        }
142
143        let generator = DefaultPackGenerator;
144        let available_pack_types = generator.available_pack_types(self);
145
146        Some(
147            available_pack_types
148                .into_iter()
149                .map(|pack_type| Action::BuyPack { pack_type }),
150        )
151    }
152
153    // Get open pack actions
154    fn gen_actions_open_pack(&self) -> Option<impl Iterator<Item = Action> + use<'_>> {
155        // Can only open packs if there are packs in inventory and no pack is currently open
156        if self.pack_inventory.is_empty() || self.open_pack.is_some() {
157            return None;
158        }
159
160        Some((0..self.pack_inventory.len()).map(|pack_id| Action::OpenPack { pack_id }))
161    }
162
163    // Get select from pack actions
164    fn gen_actions_select_from_pack(&self) -> Option<impl Iterator<Item = Action> + use<'_>> {
165        // Can only select if a pack is currently open
166        let open_pack_state = self.open_pack.as_ref()?;
167
168        Some(
169            (0..open_pack_state.pack.options.len()).map(move |option_index| {
170                Action::SelectFromPack {
171                    pack_id: open_pack_state.pack_id,
172                    option_index,
173                }
174            }),
175        )
176    }
177
178    // Get skip pack actions
179    fn gen_actions_skip_pack(&self) -> Option<impl Iterator<Item = Action> + use<'_>> {
180        // Can only skip if a pack is currently open and skippable
181        let open_pack_state = self.open_pack.as_ref()?;
182
183        if !open_pack_state.pack.can_skip {
184            return None;
185        }
186
187        Some(
188            vec![Action::SkipPack {
189                pack_id: open_pack_state.pack_id,
190            }]
191            .into_iter(),
192        )
193    }
194
195    // Get all legal actions that can be executed given current state
196    pub fn gen_actions(&self) -> impl Iterator<Item = Action> + use<'_> {
197        let select_cards = self.gen_actions_select_card();
198        let plays = self.gen_actions_play();
199        let discards = self.gen_actions_discard();
200        let move_cards = self.gen_actions_move_card();
201        let cash_outs = self.gen_actions_cash_out();
202        let next_rounds = self.gen_actions_next_round();
203        let select_blinds = self.gen_actions_select_blind();
204        let buy_jokers = self.gen_actions_buy_joker();
205        let buy_packs = self.gen_actions_buy_pack();
206        let open_packs = self.gen_actions_open_pack();
207        let select_from_packs = self.gen_actions_select_from_pack();
208        let skip_packs = self.gen_actions_skip_pack();
209
210        select_cards
211            .into_iter()
212            .flatten()
213            .chain(plays.into_iter().flatten())
214            .chain(discards.into_iter().flatten())
215            .chain(move_cards.into_iter().flatten())
216            .chain(cash_outs.into_iter().flatten())
217            .chain(next_rounds.into_iter().flatten())
218            .chain(select_blinds.into_iter().flatten())
219            .chain(buy_jokers.into_iter().flatten())
220            .chain(buy_packs.into_iter().flatten())
221            .chain(open_packs.into_iter().flatten())
222            .chain(select_from_packs.into_iter().flatten())
223            .chain(skip_packs.into_iter().flatten())
224    }

226    fn unmask_action_space_select_cards(&self, space: &mut ActionSpace) {
227        if !self.stage.is_blind() {
228            return;
229        }
230        // Cannot select more if max already selected
231        if self.available.selected().len() >= self.config.selected_max {
232            return;
233        }
234        self.available
235            .cards_and_selected()
236            .iter()
237            .enumerate()
238            .filter(|(_, (_, a))| !*a)
239            .for_each(|(i, _)| {
240                let _ = space.unmask_select_card(i); // Ignore error - defensive programming
241            });
242    }
243
244    fn unmask_action_space_play_and_discard(&self, space: &mut ActionSpace) {
245        if !self.stage.is_blind() {
246            return;
247        }
248        // Cannot play/discard if no cards selected
249        if self.available.selected().is_empty() {
250            return;
251        }
252        // Can only play/discard is have remaining
253        if self.plays != 0.0 {
254            space.unmask_play();
255        }
256        if self.discards != 0.0 {
257            space.unmask_discard();
258        }
259    }
260
261    fn unmask_action_space_move_cards(&self, space: &mut ActionSpace) {
262        if !self.stage.is_blind() {
263            return;
264        }
265        // move left
266        // every available card except the first can move left
267        self.available
268            .cards()
269            .iter()
270            .skip(1)
271            .enumerate()
272            .for_each(|(i, _)| {
273                let _ = space.unmask_move_card_left(i); // Ignore error - defensive programming
274            });
275        // move right
276        // every available card except the last can move right
277        self.available
278            .cards()
279            .iter()
280            .rev()
281            .skip(1)
282            .rev()
283            .enumerate()
284            .for_each(|(i, _)| {
285                let _ = space.unmask_move_card_right(i); // Ignore error - defensive programming
286            });
287    }
288
289    fn unmask_action_space_cash_out(&self, space: &mut ActionSpace) {
290        if self.stage != Stage::PostBlind() {
291            return;
292        }
293        space.unmask_cash_out();
294    }
295
296    fn unmask_action_space_next_round(&self, space: &mut ActionSpace) {
297        if self.stage != Stage::Shop() {
298            return;
299        }
300        space.unmask_next_round();
301    }
302
303    fn unmask_action_space_select_blind(&self, space: &mut ActionSpace) {
304        if self.stage != Stage::PreBlind() {
305            return;
306        }
307        space.unmask_select_blind();
308    }
309
310    fn unmask_action_space_buy_joker(&self, space: &mut ActionSpace) {
311        if self.stage != Stage::Shop() {
312            return;
313        }
314        self.shop
315            .jokers
316            .iter()
317            .enumerate()
318            .filter(|(_i, j)| j.cost() as f64 <= self.money)
319            .for_each(|(i, _j)| {
320                let _ = space.unmask_buy_joker(i); // Ignore error - defensive programming
321            });
322    }
323
324    // Get an action space, masked for legal actions only
325    pub fn gen_action_space(&self) -> ActionSpace {
326        let mut space = ActionSpace::from(self.config.clone());
327        self.unmask_action_space_select_cards(&mut space);
328        self.unmask_action_space_play_and_discard(&mut space);
329        self.unmask_action_space_move_cards(&mut space);
330        self.unmask_action_space_cash_out(&mut space);
331        self.unmask_action_space_next_round(&mut space);
332        self.unmask_action_space_select_blind(&mut space);
333        self.unmask_action_space_buy_joker(&mut space);
334        space
335    }

489        let available = g.available.cards().len();
490

493        let move_range_size = safe_size_for_move_operations(available);

497        }
498        for i in 0..move_range_size {
499            assert!(space.move_card_right[i] == 1);

506        }
507
508        // Get fresh action space and mask

897        let ace = Card::new(Value::Ace, Suit::Heart);

899
900        let actions: Vec<Action> = g.gen_actions().collect();
901
902        // Should have select action for the one card

904            .iter()
905            .any(|a| matches!(a, Action::SelectCard(c) if *c == ace)));
906
907        // Should have no move actions (only one card)
908        assert!(!actions.iter().any(|a| matches!(a, Action::MoveCard(_, _))));
909
910        // Test with maximum available cards (config limit)
911        // Clear existing cards first
912        g.available = crate::available::Available::default();
913        let max_cards = g.config.available_max;
914        for i in 0..max_cards.min(52) {
915            // Don't exceed standard deck size
916            let value = Value::values()[i % 13];
917            let suit = [Suit::Heart, Suit::Diamond, Suit::Club, Suit::Spade][i % 4];
918            g.available.extend(vec![Card::new(value, suit)]);
919        }
920
921        let actions: Vec<Action> = g.gen_actions().collect();
922
923        // Should have select actions for available cards
924        let select_actions: Vec<&Action> = actions
925            .iter()
926            .filter(|a| matches!(a, Action::SelectCard(_)))
927            .collect();
