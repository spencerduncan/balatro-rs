1//! Vouchers module for Balatro game engine
2//!
3//! This module provides the infrastructure for voucher cards in Balatro.
4//! Vouchers are permanent upgrades that can be purchased in the shop.
5//!
6//! # Module Organization
7//!
8//! - `mod.rs` - Core types and traits for vouchers
9//! - `implementations.rs` - Specific voucher implementations (future)
10//!
11//! # Design Principles
12//!
13//! - Vouchers provide permanent effects that persist across rounds
14//! - Each voucher can only be purchased once per run
15//! - Vouchers may have prerequisites (other vouchers that must be owned first)
16//! - Effects are applied passively to game state
17

28/// Errors that can occur during voucher operations
29#[derive(Error, Debug, Clone)]
30pub enum VoucherError {
31    #[error("Hand size increase too large: {amount} (max: 50)")]
32    ExcessiveHandSize { amount: usize },
33    #[error("Money gain too large: {amount} (max: 10000)")]
34    ExcessiveMoneyGain { amount: usize },
35    #[error("Invalid ante scaling: {multiplier} (must be finite, positive, and ≤ 10.0)")]
36    InvalidScaling { multiplier: f64 },
37    #[error("Too many pack options: {amount} (max: 10)")]
38    ExcessivePackOptions { amount: usize },
39    #[error("Invalid blind score reduction: {multiplier} (must be finite, positive, and ≤ 1.0)")]
40    InvalidBlindReduction { multiplier: f64 },
41    #[error("Too many starting cards: {count} (max: 52)")]
42    ExcessiveStartingCards { count: usize },
43    #[error("Shop slot increase too large: {amount} (max: 20)")]
44    ExcessiveShopSlots { amount: usize },
45    #[error("Discard increase too large: {amount} (max: 50)")]
46    ExcessiveDiscards { amount: usize },
47    #[error("Play increase too large: {amount} (max: 50)")]
48    ExcessivePlays { amount: usize },
49    #[error("Joker slot increase too large: {amount} (max: 20)")]
50    ExcessiveJokerSlots { amount: usize },
51}

62/// Categorization of voucher effects for game system integration
63#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
64pub enum VoucherEffect {
65    /// Increases hand size by the specified amount
66    HandSizeIncrease(usize),
67    /// Decreases hand size by the specified amount (for negative effects)
68    HandSizeDecrease(usize),
69    /// Increases joker slots by the specified amount
70    JokerSlotIncrease(usize),
71    /// Decreases joker slots by the specified amount (for negative effects)
72    JokerSlotDecrease(usize),
73    /// Provides money gain (immediate or per-round)
74    MoneyGain(usize),
75    /// Increases the interest cap by the specified amount
76    InterestCapIncrease(usize),
77    /// Modifies ante scaling (multiplier)
78    AnteScaling(f64),
79    /// Increases ante required to win by the specified amount
80    AnteWinRequirementIncrease(usize),
81    /// Adds extra pack options in shop
82    ExtraPackOptions(usize),
83    /// Reduces blind score requirements (multiplier)
84    BlindScoreReduction(f64),
85    /// Adds starting cards to deck
86    StartingCards(Vec<crate::card::Card>),
87    /// Increases shop slots
88    ShopSlotIncrease(usize),
89    /// Increases discards per round
90    DiscardIncrease(usize),
91    /// Decreases discards per round (for negative effects)
92    DiscardDecrease(usize),
93    /// Increases plays per round
94    PlayIncrease(usize),
95    /// Enables playing cards to be purchased from shop
96    ShopPlayingCardsEnabled,
97    /// Enables playing cards in shop to have enhancements
98    ShopEnhancementsEnabled,
99    /// Multiplies Tarot card appearance frequency
100    TarotFrequencyMultiplier(f64),
101    /// No effect (flavor voucher)
102    NoEffect,
103}

105impl VoucherEffect {
106    /// Check if this effect is permanent (applies for entire run)
107    pub fn is_permanent(&self) -> bool {
108        match self {
109            VoucherEffect::MoneyGain(_) => false, // One-time effect
110            _ => true,                            // Most effects are permanent
111        }
112    }
113
114    /// Check if this effect affects shop mechanics
115    /// Check if this effect affects shop mechanics
116    pub fn affects_shop(&self) -> bool {
117        matches!(
118            self,
119            VoucherEffect::ExtraPackOptions(_)
120                | VoucherEffect::ShopSlotIncrease(_)
121                | VoucherEffect::JokerSlotIncrease(_)
122                | VoucherEffect::JokerSlotDecrease(_)
123                | VoucherEffect::ShopPlayingCardsEnabled
124                | VoucherEffect::ShopEnhancementsEnabled
125        )
126    }
127
128    /// Check if this effect affects money systems
129    pub fn affects_money(&self) -> bool {
130        matches!(self, VoucherEffect::MoneyGain(_))
131    }
132
133    /// Check if this effect affects hand mechanics
134    pub fn affects_hand(&self) -> bool {
135        matches!(
136            self,
137            VoucherEffect::HandSizeIncrease(_)
138                | VoucherEffect::DiscardIncrease(_)
139                | VoucherEffect::PlayIncrease(_)
140        )
141    }
142
143    /// Check if this effect has a numeric value
144    pub fn has_numeric_value(&self) -> bool {
145        !matches!(self, VoucherEffect::StartingCards(_))
146    }
147
148    /// Get hand size bonus if applicable
149    pub fn hand_size_bonus(&self) -> Option<usize> {
150        match self {
151            VoucherEffect::HandSizeIncrease(amount) => Some(*amount),
152            _ => None,
153        }
154    }
155
156    /// Get joker slot bonus if applicable
157    pub fn joker_slot_bonus(&self) -> Option<usize> {
158        match self {
159            VoucherEffect::JokerSlotIncrease(amount) => Some(*amount),
160            _ => None,
161        }
162    }
163
164    /// Get money bonus if applicable
165    pub fn money_bonus(&self) -> Option<usize> {
166        match self {
167            VoucherEffect::MoneyGain(amount) => Some(*amount),
168            _ => None,
169        }
170    }
171
172    /// Validate that the effect has reasonable bounds
173    pub fn validate(&self) -> Result<(), VoucherError> {
174        match self {
175            VoucherEffect::HandSizeIncrease(amount) => {
176                if *amount > 50 {
177                    return Err(VoucherError::ExcessiveHandSize { amount: *amount });
178                }
179            }
180            VoucherEffect::JokerSlotIncrease(amount) => {
181                if *amount > 20 {
182                    return Err(VoucherError::ExcessiveJokerSlots { amount: *amount });
183                }
184            }
185            VoucherEffect::MoneyGain(amount) => {
186                if *amount > 10000 {
187                    return Err(VoucherError::ExcessiveMoneyGain { amount: *amount });
188                }
189            }
190            VoucherEffect::AnteScaling(multiplier) => {
191                if !multiplier.is_finite() || *multiplier <= 0.0 || *multiplier > 10.0 {
192                    return Err(VoucherError::InvalidScaling {
193                        multiplier: *multiplier,
194                    });
195                }
196            }
197            VoucherEffect::ExtraPackOptions(amount) => {
198                if *amount > 10 {
199                    return Err(VoucherError::ExcessivePackOptions { amount: *amount });
200                }
201            }
202            VoucherEffect::BlindScoreReduction(multiplier) => {
203                if !multiplier.is_finite() || *multiplier <= 0.0 || *multiplier > 1.0 {
204                    return Err(VoucherError::InvalidBlindReduction {
205                        multiplier: *multiplier,
206                    });
207                }
208            }
209            VoucherEffect::StartingCards(cards) => {
210                if cards.len() > 52 {
211                    return Err(VoucherError::ExcessiveStartingCards { count: cards.len() });
212                }
213            }
214            VoucherEffect::ShopSlotIncrease(amount) => {
215                if *amount > 20 {
216                    return Err(VoucherError::ExcessiveShopSlots { amount: *amount });
217                }
218            }
219            VoucherEffect::DiscardIncrease(amount) => {
220                if *amount > 50 {
221                    return Err(VoucherError::ExcessiveDiscards { amount: *amount });
222                }
223            }
224            VoucherEffect::PlayIncrease(amount) => {
225                if *amount > 50 {
226                    return Err(VoucherError::ExcessivePlays { amount: *amount });
227                }
228            }
229            VoucherEffect::HandSizeDecrease(amount) => {
230                if *amount > 50 {
231                    return Err(VoucherError::ExcessiveHandSize { amount: *amount });
232                }
233            }
234            VoucherEffect::JokerSlotDecrease(amount) => {
235                if *amount > 20 {
236                    return Err(VoucherError::ExcessiveJokerSlots { amount: *amount });
237                }
238            }
239            VoucherEffect::InterestCapIncrease(amount) => {
240                if *amount > 10 {
241                    return Err(VoucherError::ExcessiveMoneyGain { amount: *amount });
242                }
243            }
244            VoucherEffect::AnteWinRequirementIncrease(amount) => {
245                if *amount > 8 {
246                    return Err(VoucherError::ExcessiveHandSize { amount: *amount });
247                    // Reuse error type
248                }
249            }
250            VoucherEffect::DiscardDecrease(amount) => {
251                if *amount > 50 {
252                    return Err(VoucherError::ExcessiveDiscards { amount: *amount });
253                }
254            }
255            VoucherEffect::TarotFrequencyMultiplier(multiplier) => {
256                if !multiplier.is_finite() || *multiplier <= 0.0 || *multiplier > 10.0 {
257                    return Err(VoucherError::InvalidScaling {
258                        multiplier: *multiplier,
259                    });
260                }
261            }
262            VoucherEffect::ShopPlayingCardsEnabled => {}
263            VoucherEffect::ShopEnhancementsEnabled => {}
264            VoucherEffect::NoEffect => {}
265        }
266        Ok(())
267    }

299/// Rules for how voucher effects can stack
300#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
301pub enum StackingRule {
302    /// Cannot stack - only one instance allowed
303    NoStacking,
304    /// Can stack without limit
305    UnlimitedStacking,
306    /// Can stack up to specified limit
307    LimitedStacking(usize),
308}
309
310impl StackingRule {
311    /// Check if this rule allows stacking
312    pub fn allows_stacking(&self) -> bool {
313        !matches!(self, StackingRule::NoStacking)
314    }
315
316    /// Get maximum stack size if limited
317    pub fn max_stack_size(&self) -> Option<usize> {
318        match self {
319            StackingRule::NoStacking => Some(1),
320            StackingRule::UnlimitedStacking => None,
321            StackingRule::LimitedStacking(limit) => Some(*limit),
322        }
323    }
324
325    /// Check if this stacking rule is compatible with another
326    pub fn is_compatible_with(&self, other: &StackingRule) -> bool {
327        self == other
328    }
329}

395    /// Apply a voucher effect to the game state with validation and safety checks
396    pub fn apply_voucher_effect(&mut self, effect: &VoucherEffect) -> Result<(), GameStateError> {
397        // First validate the effect
398        effect.validate()?;
399
400        // Apply the effect with bounds checking
401        match effect {
402            VoucherEffect::HandSizeIncrease(amount) => {
403                self.hand_size = (self.hand_size + amount).min(50);
404            }
405            VoucherEffect::JokerSlotIncrease(amount) => {
406                self.joker_slots = (self.joker_slots + amount).min(20);
407            }
408            VoucherEffect::MoneyGain(amount) => {
409                self.money = self.money.saturating_add(*amount);
410            }
411            VoucherEffect::AnteScaling(_multiplier) => {
412                // Ante scaling affects ante progression, not current ante
413                // This would be handled by the game engine during ante advancement
414            }
415            VoucherEffect::ExtraPackOptions(_amount) => {
416                // Pack options affect shop generation, not game state directly
417                // This would be handled by the shop system
418            }
419            VoucherEffect::BlindScoreReduction(_multiplier) => {
420                // Blind score reduction affects scoring calculations, not game state directly
421                // This would be handled by the scoring system
422            }
423            VoucherEffect::StartingCards(_cards) => {
424                // Starting cards affect deck initialization, not current game state
425                // This would be handled during game setup
426            }
427            VoucherEffect::ShopSlotIncrease(_amount) => {
428                // Shop slots affect shop generation, not game state directly
429                // This would be handled by the shop system
430            }
431            VoucherEffect::DiscardIncrease(_amount) => {
432                // Discard increases affect round mechanics, not persistent game state
433                // This would be handled by the round system
434            }
435            VoucherEffect::PlayIncrease(_amount) => {
436                // Play increases affect round mechanics, not persistent game state
437                // This would be handled by the round system
438            }
439            VoucherEffect::HandSizeDecrease(amount) => {
440                self.hand_size = self.hand_size.saturating_sub(*amount).max(1);
441            }
442            VoucherEffect::JokerSlotDecrease(amount) => {
443                self.joker_slots = self.joker_slots.saturating_sub(*amount).max(1);
444            }
445            VoucherEffect::InterestCapIncrease(_amount) => {
446                // Interest cap increases affect interest calculation, not game state directly
447                // This would be handled by the interest system
448            }
449            VoucherEffect::AnteWinRequirementIncrease(_amount) => {
450                // Ante win requirement affects victory condition, not current game state
451                // This would be handled by the victory system
452            }
453            VoucherEffect::DiscardDecrease(_amount) => {
454                // Discard decreases affect round mechanics, not persistent game state
455                // This would be handled by the round system
456            }
457            VoucherEffect::TarotFrequencyMultiplier(_multiplier) => {
458                // Tarot frequency affects shop/pack generation, not game state directly
459                // This would be handled by the shop system
460            }
461            VoucherEffect::ShopPlayingCardsEnabled => {
462                // Shop playing cards enable affects shop generation, not game state directly
463                // This would be handled by the shop system
464            }
465            VoucherEffect::ShopEnhancementsEnabled => {
466                // Shop enhancements enable affects shop generation, not game state directly
467                // This would be handled by the shop system
468            }
469            VoucherEffect::NoEffect => {
470                // Blank voucher does nothing
471            }
472        }
473        // Validate final state consistency
474        self.validate_state()
475    }
476
477    /// Add a voucher to the owned collection
478    pub fn add_voucher(&mut self, voucher_id: VoucherId) {
479        self.vouchers_owned.insert(voucher_id);
480    }
481
482    /// Spend money if sufficient funds available
483    pub fn spend_money(&mut self, amount: usize) -> Result<(), GameStateError> {
484        if self.money < amount {
485            return Err(GameStateError::InvalidState {
486                reason: format!("Insufficient funds: have {}, need {}", self.money, amount),
487            });
488        }
489        self.money -= amount;
490        Ok(())
491    }
492
493    /// Validate that the game state is consistent and within reasonable bounds
494    pub fn validate_state(&self) -> Result<(), GameStateError> {
495        if self.hand_size > 50 {
496            return Err(GameStateError::InvalidState {
497                reason: format!("Hand size too large: {}", self.hand_size),
498            });
499        }
500        if self.joker_slots > 20 {
501            return Err(GameStateError::InvalidState {
502                reason: format!("Too many joker slots: {}", self.joker_slots),
503            });
504        }
505        if self.ante > 8 {
506            return Err(GameStateError::InvalidState {
507                reason: format!("Ante too high: {}", self.ante),
508            });
509        }
510        if self.vouchers_owned.len() > 100 {
511            return Err(GameStateError::InvalidState {
512                reason: format!("Too many vouchers owned: {}", self.vouchers_owned.len()),
513            });
514        }
515        Ok(())
516    }
517}

519impl From<&crate::game::Game> for GameState {
520    fn from(game: &crate::game::Game) -> Self {
521        // Convert Ante enum to usize
522        let ante_value = match game.ante_current {
523            crate::ante::Ante::Zero => 0,
524            crate::ante::Ante::One => 1,
525            crate::ante::Ante::Two => 2,
526            crate::ante::Ante::Three => 3,
527            crate::ante::Ante::Four => 4,
528            crate::ante::Ante::Five => 5,
529            crate::ante::Ante::Six => 6,
530            crate::ante::Ante::Seven => 7,
531            crate::ante::Ante::Eight => 8,
532        };
533
534        Self {
535            money: game.money as usize,
536            ante: ante_value,
537            hand_size: 8, // Base hand size, vouchers would modify this
538            joker_slots: game.config.joker_slots,
539            vouchers_owned: game.vouchers.owned_vouchers().into_iter().collect(),
540        }
541    }
542}

776/// Set of vouchers owned by the player
777/// Provides efficient lookup and management of owned vouchers
778#[derive(Debug, Clone, Default, Serialize, Deserialize)]
779pub struct VoucherCollection {
780    owned: std::collections::HashSet<VoucherId>,
781}
782
783impl VoucherCollection {
784    /// Create a new empty voucher collection
785    pub fn new() -> Self {
786        Self {
787            owned: std::collections::HashSet::new(),
788        }
789    }
790
791    /// Add a voucher to the collection
792    pub fn add(&mut self, voucher: VoucherId) {
793        self.owned.insert(voucher);
794    }
795
796    /// Check if a voucher is owned
797    pub fn owns(&self, voucher: VoucherId) -> bool {
798        self.owned.contains(&voucher)
799    }
800
801    /// Get all owned vouchers
802    pub fn owned_vouchers(&self) -> Vec<VoucherId> {
803        self.owned.iter().copied().collect()
804    }
805
806    /// Check if all prerequisites for a voucher are met
807    pub fn can_purchase(&self, voucher: VoucherId) -> bool {
808        if self.owns(voucher) {
809            return false; // Already owned
810        }
811
812        voucher
813            .prerequisites()
814            .iter()
815            .all(|&prereq| self.owns(prereq))
816    }
817
818    /// Get the number of vouchers owned
819    pub fn count(&self) -> usize {
820        self.owned.len()
821    }
822}
