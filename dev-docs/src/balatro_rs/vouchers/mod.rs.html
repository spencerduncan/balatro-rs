1//! Vouchers module for Balatro game engine
2//!
3//! This module provides the infrastructure for voucher cards in Balatro.
4//! Vouchers are permanent upgrades that can be purchased in the shop.
5//!
6//! # Module Organization
7//!
8//! - `mod.rs` - Core types and traits for vouchers
9//! - `implementations.rs` - Specific voucher implementations (future)
10//!
11//! # Design Principles
12//!
13//! - Vouchers provide permanent effects that persist across rounds
14//! - Each voucher can only be purchased once per run
15//! - Vouchers may have prerequisites (other vouchers that must be owned first)
16//! - Effects are applied passively to game state
17

28/// Errors that can occur during voucher operations
29#[derive(Error, Debug, Clone)]
30pub enum VoucherError {
31    #[error("Hand size increase too large: {amount} (max: 50)")]
32    ExcessiveHandSize { amount: usize },
33    #[error("Money gain too large: {amount} (max: 10000)")]
34    ExcessiveMoneyGain { amount: usize },
35    #[error("Invalid ante scaling: {multiplier} (must be finite, positive, and ≤ 10.0)")]
36    InvalidScaling { multiplier: f64 },
37    #[error("Too many pack options: {amount} (max: 10)")]
38    ExcessivePackOptions { amount: usize },
39    #[error("Invalid blind score reduction: {multiplier} (must be finite, positive, and ≤ 1.0)")]
40    InvalidBlindReduction { multiplier: f64 },
41    #[error("Too many starting cards: {count} (max: 52)")]
42    ExcessiveStartingCards { count: usize },
43    #[error("Shop slot increase too large: {amount} (max: 20)")]
44    ExcessiveShopSlots { amount: usize },
45    #[error("Discard increase too large: {amount} (max: 50)")]
46    ExcessiveDiscards { amount: usize },
47    #[error("Play increase too large: {amount} (max: 50)")]
48    ExcessivePlays { amount: usize },
49    #[error("Joker slot increase too large: {amount} (max: 20)")]
50    ExcessiveJokerSlots { amount: usize },
51}

62/// Categorization of voucher effects for game system integration
63#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
64pub enum VoucherEffect {
65    /// Increases hand size by the specified amount
66    HandSizeIncrease(usize),
67    /// Decreases hand size by the specified amount (for negative effects)
68    HandSizeDecrease(usize),
69    /// Increases joker slots by the specified amount
70    JokerSlotIncrease(usize),
71    /// Decreases joker slots by the specified amount (for negative effects)
72    JokerSlotDecrease(usize),
73    /// Provides money gain (immediate or per-round)
74    MoneyGain(usize),
75    /// Increases the interest cap by the specified amount
76    InterestCapIncrease(usize),
77    /// Modifies ante scaling (multiplier)
78    AnteScaling(f64),
79    /// Increases ante required to win by the specified amount
80    AnteWinRequirementIncrease(usize),
81    /// Decreases ante required to win by the specified amount
82    AnteWinRequirementDecrease(usize),
83    /// Adds extra pack options in shop
84    ExtraPackOptions(usize),
85    /// Reduces blind score requirements (multiplier)
86    BlindScoreReduction(f64),
87    /// Adds starting cards to deck
88    StartingCards(Vec<crate::card::Card>),
89    /// Increases shop slots
90    ShopSlotIncrease(usize),
91    /// Increases discards per round
92    DiscardIncrease(usize),
93    /// Decreases discards per round (for negative effects)
94    DiscardDecrease(usize),
95    /// Increases plays per round
96    PlayIncrease(usize),
97    /// Enables playing cards to be purchased from shop
98    ShopPlayingCardsEnabled,
99    /// Enables playing cards in shop to have enhancements
100    ShopEnhancementsEnabled,
101    /// Multiplies Tarot card appearance frequency
102    TarotFrequencyMultiplier(f64),
103    /// Multiplies Planet card appearance frequency
104    PlanetFrequencyMultiplier(f64),
105    /// Multiplies enhanced card (foil/holo/polychrome) appearance frequency
106    PolychromeFrequencyMultiplier(f64),
107    /// Provides percentage discount on all shop items
108    ShopDiscountPercent(f64),
109    /// Applies discount multiplier to shop items (0.5 = 50% off)
110    ShopDiscountMultiplier(f64),
111    /// Reduces reroll cost by specified amount
112    RerollCostReduction(usize),
113    /// Increases consumable slots
114    ConsumableSlotIncrease(usize),
115    /// Enables boss blind reroll functionality (limited or unlimited)
116    BossBlindRerollEnabled {
117        unlimited: bool,
118        cost_per_roll: usize,
119    },
120    /// No effect (flavor voucher)
121    NoEffect,
122}

199    /// Validate that the effect has reasonable bounds
200    pub fn validate(&self) -> Result<(), VoucherError> {
201        match self {
202            VoucherEffect::HandSizeIncrease(amount) => {
203                if *amount > 50 {
204                    return Err(VoucherError::ExcessiveHandSize { amount: *amount });
205                }
206            }
207            VoucherEffect::JokerSlotIncrease(amount) => {
208                if *amount > 20 {
209                    return Err(VoucherError::ExcessiveJokerSlots { amount: *amount });
210                }
211            }
212            VoucherEffect::MoneyGain(amount) => {
213                if *amount > 10000 {
214                    return Err(VoucherError::ExcessiveMoneyGain { amount: *amount });
215                }
216            }
217            VoucherEffect::AnteScaling(multiplier) => {
218                if !multiplier.is_finite() || *multiplier <= 0.0 || *multiplier > 10.0 {
219                    return Err(VoucherError::InvalidScaling {
220                        multiplier: *multiplier,
221                    });
222                }
223            }
224            VoucherEffect::ExtraPackOptions(amount) => {
225                if *amount > 10 {
226                    return Err(VoucherError::ExcessivePackOptions { amount: *amount });
227                }
228            }
229            VoucherEffect::BlindScoreReduction(multiplier) => {
230                if !multiplier.is_finite() || *multiplier <= 0.0 || *multiplier > 1.0 {
231                    return Err(VoucherError::InvalidBlindReduction {
232                        multiplier: *multiplier,
233                    });
234                }
235            }
236            VoucherEffect::StartingCards(cards) => {
237                if cards.len() > 52 {
238                    return Err(VoucherError::ExcessiveStartingCards { count: cards.len() });
239                }
240            }
241            VoucherEffect::ShopSlotIncrease(amount) => {
242                if *amount > 20 {
243                    return Err(VoucherError::ExcessiveShopSlots { amount: *amount });
244                }
245            }
246            VoucherEffect::DiscardIncrease(amount) => {
247                if *amount > 50 {
248                    return Err(VoucherError::ExcessiveDiscards { amount: *amount });
249                }
250            }
251            VoucherEffect::PlayIncrease(amount) => {
252                if *amount > 50 {
253                    return Err(VoucherError::ExcessivePlays { amount: *amount });
254                }
255            }
256            VoucherEffect::HandSizeDecrease(amount) => {
257                if *amount > 50 {
258                    return Err(VoucherError::ExcessiveHandSize { amount: *amount });
259                }
260            }
261            VoucherEffect::JokerSlotDecrease(amount) => {
262                if *amount > 20 {
263                    return Err(VoucherError::ExcessiveJokerSlots { amount: *amount });
264                }
265            }
266            VoucherEffect::InterestCapIncrease(amount) => {
267                if *amount > 10 {
268                    return Err(VoucherError::ExcessiveMoneyGain { amount: *amount });
269                }
270            }
271            VoucherEffect::AnteWinRequirementIncrease(amount) => {
272                if *amount > 8 {
273                    return Err(VoucherError::ExcessiveHandSize { amount: *amount });
274                    // Reuse error type
275                }
276            }
277            VoucherEffect::AnteWinRequirementDecrease(amount) => {
278                if *amount > 8 {
279                    return Err(VoucherError::ExcessiveHandSize { amount: *amount });
280                    // Reuse error type - same bounds as increase
281                }
282            }
283            VoucherEffect::DiscardDecrease(amount) => {
284                if *amount > 50 {
285                    return Err(VoucherError::ExcessiveDiscards { amount: *amount });
286                }
287            }
288            VoucherEffect::TarotFrequencyMultiplier(multiplier) => {
289                if !multiplier.is_finite() || *multiplier <= 0.0 || *multiplier > 10.0 {
290                    return Err(VoucherError::InvalidScaling {
291                        multiplier: *multiplier,
292                    });
293                }
294            }
295            VoucherEffect::PlanetFrequencyMultiplier(multiplier) => {
296                if !multiplier.is_finite() || *multiplier <= 0.0 || *multiplier > 10.0 {
297                    return Err(VoucherError::InvalidScaling {
298                        multiplier: *multiplier,
299                    });
300                }
301            }
302            VoucherEffect::PolychromeFrequencyMultiplier(multiplier) => {
303                if !multiplier.is_finite() || *multiplier <= 0.0 || *multiplier > 10.0 {
304                    return Err(VoucherError::InvalidScaling {
305                        multiplier: *multiplier,
306                    });
307                }
308            }
309            VoucherEffect::ShopDiscountPercent(discount) => {
310                if !discount.is_finite() || *discount <= 0.0 || *discount > 100.0 {
311                    return Err(VoucherError::InvalidScaling {
312                        multiplier: *discount,
313                    });
314                }
315            }
316            VoucherEffect::ShopDiscountMultiplier(multiplier) => {
317                if !multiplier.is_finite() || *multiplier <= 0.0 || *multiplier > 1.0 {
318                    return Err(VoucherError::InvalidBlindReduction {
319                        multiplier: *multiplier,
320                    });
321                }
322            }
323            VoucherEffect::RerollCostReduction(amount) => {
324                if *amount > 10 {
325                    return Err(VoucherError::ExcessiveMoneyGain { amount: *amount });
326                }
327            }
328            VoucherEffect::ConsumableSlotIncrease(amount) => {
329                if *amount > 10 {
330                    return Err(VoucherError::ExcessiveJokerSlots { amount: *amount });
331                }
332            }
333            VoucherEffect::BossBlindRerollEnabled { cost_per_roll, .. } => {
334                if *cost_per_roll > 100 {
335                    return Err(VoucherError::ExcessiveMoneyGain {
336                        amount: *cost_per_roll,
337                    });
338                }
339            }
340            VoucherEffect::ShopPlayingCardsEnabled => {}
341            VoucherEffect::ShopEnhancementsEnabled => {}
342            VoucherEffect::NoEffect => {}
343        }
344        Ok(())
345    }

377/// Rules for how voucher effects can stack
378#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
379pub enum StackingRule {
380    /// Cannot stack - only one instance allowed
381    NoStacking,
382    /// Can stack without limit
383    UnlimitedStacking,
384    /// Can stack up to specified limit
385    LimitedStacking(usize),
386}
387
388impl StackingRule {
389    /// Check if this rule allows stacking
390    pub fn allows_stacking(&self) -> bool {
391        !matches!(self, StackingRule::NoStacking)
392    }
393
394    /// Get maximum stack size if limited
395    pub fn max_stack_size(&self) -> Option<usize> {
396        match self {
397            StackingRule::NoStacking => Some(1),
398            StackingRule::UnlimitedStacking => None,
399            StackingRule::LimitedStacking(limit) => Some(*limit),
400        }
401    }
402
403    /// Check if this stacking rule is compatible with another
404    pub fn is_compatible_with(&self, other: &StackingRule) -> bool {
405        self == other
406    }
407}

473    /// Apply a voucher effect to the game state with validation and safety checks
474    pub fn apply_voucher_effect(&mut self, effect: &VoucherEffect) -> Result<(), GameStateError> {
475        // First validate the effect
476        effect.validate()?;
477
478        // Apply the effect with bounds checking
479        match effect {
480            VoucherEffect::HandSizeIncrease(amount) => {
481                self.hand_size = (self.hand_size + amount).min(50);
482            }
483            VoucherEffect::JokerSlotIncrease(amount) => {
484                self.joker_slots = (self.joker_slots + amount).min(20);
485            }
486            VoucherEffect::MoneyGain(amount) => {
487                self.money = self.money.saturating_add(*amount);
488            }
489            VoucherEffect::AnteScaling(_multiplier) => {
490                // Ante scaling affects ante progression, not current ante
491                // This would be handled by the game engine during ante advancement
492            }
493            VoucherEffect::ExtraPackOptions(_amount) => {
494                // Pack options affect shop generation, not game state directly
495                // This would be handled by the shop system
496            }
497            VoucherEffect::BlindScoreReduction(_multiplier) => {
498                // Blind score reduction affects scoring calculations, not game state directly
499                // This would be handled by the scoring system
500            }
501            VoucherEffect::StartingCards(_cards) => {
502                // Starting cards affect deck initialization, not current game state
503                // This would be handled during game setup
504            }
505            VoucherEffect::ShopSlotIncrease(_amount) => {
506                // Shop slots affect shop generation, not game state directly
507                // This would be handled by the shop system
508            }
509            VoucherEffect::DiscardIncrease(_amount) => {
510                // Discard increases affect round mechanics, not persistent game state
511                // This would be handled by the round system
512            }
513            VoucherEffect::PlayIncrease(_amount) => {
514                // Play increases affect round mechanics, not persistent game state
515                // This would be handled by the round system
516            }
517            VoucherEffect::HandSizeDecrease(amount) => {
518                self.hand_size = self.hand_size.saturating_sub(*amount).max(1);
519            }
520            VoucherEffect::JokerSlotDecrease(amount) => {
521                self.joker_slots = self.joker_slots.saturating_sub(*amount).max(1);
522            }
523            VoucherEffect::InterestCapIncrease(_amount) => {
524                // Interest cap increases affect interest calculation, not game state directly
525                // This would be handled by the interest system
526            }
527            VoucherEffect::AnteWinRequirementIncrease(_amount) => {
528                // Ante win requirement affects victory condition, not current game state
529                // This would be handled by the victory system
530            }
531            VoucherEffect::AnteWinRequirementDecrease(_amount) => {
532                // Ante win requirement affects victory condition, not current game state
533                // This would be handled by the victory system
534            }
535            VoucherEffect::DiscardDecrease(_amount) => {
536                // Discard decreases affect round mechanics, not persistent game state
537                // This would be handled by the round system
538            }
539            VoucherEffect::TarotFrequencyMultiplier(_multiplier) => {
540                // Tarot frequency affects shop/pack generation, not game state directly
541                // This would be handled by the shop system
542            }
543            VoucherEffect::PlanetFrequencyMultiplier(_multiplier) => {
544                // Planet frequency affects shop/pack generation, not game state directly

548                // Enhanced card frequency affects shop/pack generation, not game state directly
549                // This would be handled by the shop system

552                // Shop discount affects item pricing, not game state directly
553                // This would be handled by the shop system
554            }
555            VoucherEffect::ShopDiscountMultiplier(_multiplier) => {
556                // Shop discount affects shop pricing, not game state directly

558            }
559            VoucherEffect::RerollCostReduction(_amount) => {
560                // Reroll cost reduction affects shop reroll pricing, not game state directly
561                // This would be handled by the shop system
562            }
563            VoucherEffect::ConsumableSlotIncrease(_amount) => {
564                // Consumable slots affect inventory capacity, not current game state

592    /// Spend money if sufficient funds available
593    pub fn spend_money(&mut self, amount: usize) -> Result<(), GameStateError> {
594        if self.money < amount {
595            return Err(GameStateError::InvalidState {
596                reason: format!("Insufficient funds: have {}, need {}", self.money, amount),
597            });
598        }
599        self.money -= amount;
600        Ok(())
601    }
602
603    /// Validate that the game state is consistent and within reasonable bounds
604    pub fn validate_state(&self) -> Result<(), GameStateError> {
605        if self.hand_size > 50 {
606            return Err(GameStateError::InvalidState {
607                reason: format!("Hand size too large: {}", self.hand_size),
608            });
609        }
610        if self.joker_slots > 20 {
611            return Err(GameStateError::InvalidState {
612                reason: format!("Too many joker slots: {}", self.joker_slots),
613            });
614        }
615        if self.ante > 8 {
616            return Err(GameStateError::InvalidState {
617                reason: format!("Ante too high: {}", self.ante),
618            });
619        }
620        if self.vouchers_owned.len() > 100 {
621            return Err(GameStateError::InvalidState {
622                reason: format!("Too many vouchers owned: {}", self.vouchers_owned.len()),
623            });
624        }
625        Ok(())
626    }

629impl From<&crate::game::Game> for GameState {
630    fn from(game: &crate::game::Game) -> Self {
631        // Convert Ante enum to usize
632        let ante_value = match game.ante_current {
633            crate::ante::Ante::Zero => 0,
634            crate::ante::Ante::One => 1,
635            crate::ante::Ante::Two => 2,
636            crate::ante::Ante::Three => 3,
637            crate::ante::Ante::Four => 4,
638            crate::ante::Ante::Five => 5,
639            crate::ante::Ante::Six => 6,
640            crate::ante::Ante::Seven => 7,
641            crate::ante::Ante::Eight => 8,
642        };
643
644        Self {
645            money: game.money as usize,
646            ante: ante_value,
647            hand_size: 8, // Base hand size, vouchers would modify this
648            joker_slots: game.config.joker_slots,
649            vouchers_owned: game.vouchers.owned_vouchers().into_iter().collect(),
650        }
651    }

898/// Set of vouchers owned by the player
899/// Provides efficient lookup and management of owned vouchers
900#[derive(Debug, Clone, Default, Serialize, Deserialize)]
901pub struct VoucherCollection {
902    owned: std::collections::HashSet<VoucherId>,
903}
904
905impl VoucherCollection {
906    /// Create a new empty voucher collection
907    pub fn new() -> Self {
908        Self {
909            owned: std::collections::HashSet::new(),
910        }
911    }
912
913    /// Add a voucher to the collection
914    pub fn add(&mut self, voucher: VoucherId) {
915        self.owned.insert(voucher);
916    }
917
918    /// Check if a voucher is owned
919    pub fn owns(&self, voucher: VoucherId) -> bool {
920        self.owned.contains(&voucher)
921    }
922
923    /// Get all owned vouchers
924    pub fn owned_vouchers(&self) -> Vec<VoucherId> {
925        self.owned.iter().copied().collect()
926    }
927
928    /// Check if all prerequisites for a voucher are met
929    pub fn can_purchase(&self, voucher: VoucherId) -> bool {
930        if self.owns(voucher) {
931            return false; // Already owned
932        }
933
934        voucher
935            .prerequisites()
936            .iter()
937            .all(|&prereq| self.owns(prereq))
938    }
939
940    /// Get the number of vouchers owned
941    pub fn count(&self) -> usize {
942        self.owned.len()
943    }
944}
