50impl TomlJokerParser {
51    /// Create a new parser with default configuration
52    pub fn new() -> Self {
53        let mut parser = Self {
54            _supported_versions: vec!["1.0.0".to_string()],
55            validators: Vec::new(),
56        };
57
58        // Add default validators
59        parser.add_validator(Box::new(SchemaVersionValidator));
60        parser.add_validator(Box::new(UniqueIdValidator));
61        parser.add_validator(Box::new(RequiredFieldValidator));
62        parser.add_validator(Box::new(RangeValidator));
63        parser.add_validator(Box::new(LogicValidator));
64
65        parser
66    }
67
68    /// Add a custom validator
69    pub fn add_validator(&mut self, validator: Box<dyn JokerValidator>) {
70        self.validators.push(validator);
71    }
72
73    /// Parse joker definitions from a TOML file
74    pub fn parse_file<P: AsRef<Path>>(&self, path: P) -> Result<JokerConfig, TomlParserError> {
75        let content = fs::read_to_string(path)?;
76        self.parse_string(&content)
77    }
78
79    /// Parse joker definitions from a TOML string
80    pub fn parse_string(&self, content: &str) -> Result<JokerConfig, TomlParserError> {
81        // Parse TOML
82        let config: JokerConfig = toml::from_str(content)?;
83
84        // Validate
85        self.validate_config(&config)?;
86
87        Ok(config)
88    }
89
90    /// Validate the parsed configuration
91    fn validate_config(&self, config: &JokerConfig) -> Result<(), TomlParserError> {
92        for validator in &self.validators {
93            validator.validate(config)?;
94        }
95        Ok(())
96    }
97
98    /// Parse multiple TOML files and merge them
99    pub fn parse_directory<P: AsRef<Path>>(
100        &self,
101        dir_path: P,
102    ) -> Result<JokerConfig, TomlParserError> {
103        let mut merged_config = JokerConfig {
104            schema_version: "1.0.0".to_string(),
105            jokers: Vec::new(),
106        };
107
108        let dir = fs::read_dir(dir_path)?;
109        for entry in dir {
110            let entry = entry?;
111            let path = entry.path();
112
113            if path.extension().and_then(|s| s.to_str()) == Some("toml") {
114                let config = self.parse_file(&path)?;
115
116                // Merge jokers
117                merged_config.jokers.extend(config.jokers);
118
119                // Use the latest schema version found
120                if config.schema_version > merged_config.schema_version {
121                    merged_config.schema_version = config.schema_version;
122                }
123            }
124        }
125
126        // Validate merged configuration
127        self.validate_config(&merged_config)?;
128
129        Ok(merged_config)
130    }
131
132    /// Hot-reload support: parse and return only changed jokers
133    pub fn parse_with_hot_reload(
134        &self,
135        content: &str,
136        previous_config: Option<&JokerConfig>,
137    ) -> Result<(JokerConfig, Vec<String>), TomlParserError> {
138        let new_config = self.parse_string(content)?;
139
140        let changed_joker_ids = if let Some(prev) = previous_config {
141            self.find_changed_jokers(prev, &new_config)
142        } else {
143            // All jokers are "new" if no previous config
144            new_config.jokers.iter().map(|j| j.id.clone()).collect()
145        };
146
147        Ok((new_config, changed_joker_ids))
148    }
149
150    /// Find jokers that have changed between configurations
151    fn find_changed_jokers(&self, old: &JokerConfig, new: &JokerConfig) -> Vec<String> {
152        let mut changed = Vec::new();
153
154        // Create lookup maps
155        let old_jokers: HashMap<&str, &TomlJokerDefinition> =
156            old.jokers.iter().map(|j| (j.id.as_str(), j)).collect();
157        let new_jokers: HashMap<&str, &TomlJokerDefinition> =
158            new.jokers.iter().map(|j| (j.id.as_str(), j)).collect();
159
160        // Check for new or modified jokers
161        for (id, new_joker) in &new_jokers {
162            if let Some(old_joker) = old_jokers.get(id) {
163                // Compare serialized forms to detect changes
164                if self.joker_definition_changed(old_joker, new_joker) {
165                    changed.push(id.to_string());
166                }
167            } else {
168                // New joker
169                changed.push(id.to_string());
170            }
171        }
172
173        // Check for removed jokers
174        for id in old_jokers.keys() {
175            if !new_jokers.contains_key(id) {
176                changed.push(id.to_string());
177            }
178        }
179
180        changed
181    }
182
183    /// Check if a joker definition has changed
184    fn joker_definition_changed(
185        &self,
186        old: &TomlJokerDefinition,
187        new: &TomlJokerDefinition,
188    ) -> bool {
189        // Simple approach: serialize both and compare
190        // In a production system, you might want more sophisticated diffing
191        match (serde_json::to_string(old), serde_json::to_string(new)) {
192            (Ok(old_json), Ok(new_json)) => old_json != new_json,
193            _ => true, // Assume changed if serialization fails
194        }
195    }
196}

203/// Validates schema version compatibility
204pub struct SchemaVersionValidator;
205
206impl JokerValidator for SchemaVersionValidator {
207    fn validate(&self, config: &JokerConfig) -> Result<(), TomlParserError> {
208        let supported_versions = ["1.0.0"];
209
210        if !supported_versions.contains(&config.schema_version.as_str()) {
211            return Err(TomlParserError::UnsupportedSchemaVersion(
212                config.schema_version.clone(),
213            ));
214        }
215
216        Ok(())
217    }
218}

220/// Validates that all joker IDs are unique
221pub struct UniqueIdValidator;
222
223impl JokerValidator for UniqueIdValidator {
224    fn validate(&self, config: &JokerConfig) -> Result<(), TomlParserError> {
225        let mut seen_ids = std::collections::HashSet::new();
226
227        for joker in &config.jokers {
228            if !seen_ids.insert(&joker.id) {
229                return Err(TomlParserError::DuplicateJokerId(joker.id.clone()));
230            }
231        }
232
233        Ok(())
234    }
235}

237/// Validates that all required fields are present
238pub struct RequiredFieldValidator;
239
240impl JokerValidator for RequiredFieldValidator {
241    fn validate(&self, config: &JokerConfig) -> Result<(), TomlParserError> {
242        for joker in &config.jokers {
243            // Check required fields
244            if joker.id.is_empty() {
245                return Err(TomlParserError::MissingRequiredField("id".to_string()));
246            }
247
248            if joker.name.is_empty() {
249                return Err(TomlParserError::MissingRequiredField("name".to_string()));
250            }
251
252            if joker.description.is_empty() {
253                return Err(TomlParserError::MissingRequiredField(
254                    "description".to_string(),
255                ));
256            }
257
258            // Validate ID format (should be valid enum variant name)
259            if !joker.id.chars().all(|c| c.is_alphanumeric() || c == '_') {
260                return Err(TomlParserError::InvalidJokerId(joker.id.clone()));
261            }
262        }
263
264        Ok(())
265    }
266}

268/// Validates value ranges and constraints
269pub struct RangeValidator;
270
271impl JokerValidator for RangeValidator {
272    fn validate(&self, config: &JokerConfig) -> Result<(), TomlParserError> {
273        for joker in &config.jokers {
274            // Validate cost range
275            if let Some(cost) = joker.cost {
276                if cost > 1000 {
277                    return Err(TomlParserError::InvalidFieldValue {
278                        field: "cost".to_string(),
279                        value: cost.to_string(),
280                    });
281                }
282            }
283
284            // Validate effect values based on type
285            match &joker.effect {
286                crate::joker_toml_schema::TomlJokerEffect::Scoring {
287                    chips,
288                    mult,
289                    money,
290                    mult_multiplier,
291                    ..
292                } => {
293                    if *chips < -1000 || *chips > 1000 {
294                        return Err(TomlParserError::InvalidFieldValue {
295                            field: "chips".to_string(),
296                            value: chips.to_string(),
297                        });
298                    }
299
300                    if *mult < -1000 || *mult > 1000 {
301                        return Err(TomlParserError::InvalidFieldValue {
302                            field: "mult".to_string(),
303                            value: mult.to_string(),
304                        });
305                    }
306
307                    if *money < -100 || *money > 100 {
308                        return Err(TomlParserError::InvalidFieldValue {
309                            field: "money".to_string(),
310                            value: money.to_string(),
311                        });
312                    }
313
314                    if *mult_multiplier < 0.0 || *mult_multiplier > 10.0 {
315                        return Err(TomlParserError::InvalidFieldValue {
316                            field: "mult_multiplier".to_string(),
317                            value: mult_multiplier.to_string(),
318                        });
319                    }
320                }
321                _ => {
322                    // Additional validation for other effect types can be added here
323                }
324            }
325        }
326
327        Ok(())
328    }
