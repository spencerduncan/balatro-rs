87    /// Accumulate a JokerEffect into this accumulator
88    /// Returns false if killscreen (infinity/NaN) is detected
89    pub fn accumulate_effect(&mut self, effect: &crate::joker::JokerEffect) -> bool {
90        // Calculate total iterations (1 + retriggers)
91        let iterations = 1 + effect.retrigger as i32;
92
93        // Accumulate chips, mult, and money (additive, multiplied by retriggers)
94        self.chips = self
95            .chips
96            .saturating_add(effect.chips.saturating_mul(iterations));
97        self.mult = self
98            .mult
99            .saturating_add(effect.mult.saturating_mul(iterations));
100        self.money = self
101            .money
102            .saturating_add(effect.money.saturating_mul(iterations));
103
104        // Apply mult multiplier for each iteration (multiplicative)
105        if effect.mult_multiplier != 0.0 {
106            for _ in 0..iterations {
107                self.mult_multiplier *= effect.mult_multiplier;
108            }
109        }
110
111        // Add message if present
112        if let Some(ref message) = effect.message {
113            self.messages.push(message.clone());
114        }
115
116        // Check for killscreen (infinity or NaN)
117        if self.mult_multiplier.is_infinite() || self.mult_multiplier.is_nan() {
118            return false;
119        }
120
121        true
122    }

124    /// Merge another AccumulatedEffects into this one
125    pub fn merge(&mut self, other: &AccumulatedEffects) {
126        self.chips = self.chips.saturating_add(other.chips);
127        self.mult = self.mult.saturating_add(other.mult);
128        self.money = self.money.saturating_add(other.money);
129
130        // Multiply mult multipliers
131        if other.mult_multiplier != 0.0 {
132            self.mult_multiplier *= other.mult_multiplier;
133        }
134
135        // Append messages
136        self.messages.extend_from_slice(&other.messages);
137    }

139    /// Apply the accumulated mult multiplier to the mult value
140    pub fn apply_mult_multiplier(&mut self) {
141        if self.mult_multiplier != 0.0 && self.mult_multiplier != 1.0 {
142            let multiplied = (self.mult as f64) * self.mult_multiplier;
143            // Clamp to i32 range to prevent overflow
144            self.mult = multiplied.clamp(i32::MIN as f64, i32::MAX as f64) as i32;
145        }
146    }

517        if self.plays == 0.0 {
518            return Err(GameError::NoRemainingPlays);
519        }
520        self.plays -= 1.0;

522        let best = selected.best_hand()?;

531        self.draw(removed);

557    pub fn calc_score(&mut self, hand: MadeHand) -> f64 {
558        // compute chips and mult from hand level
559        self.chips += hand.rank.level().chips as f64;
560        self.mult += hand.rank.level().mult as f64;
561
562        // add chips for each played card
563        let card_chips: f64 = hand.hand.cards().iter().map(|c| c.chips() as f64).sum();
564        self.chips += card_chips;
565
566        // Apply JokerEffect from structured joker system
567        if !self.jokers.is_empty() {
568            let (joker_chips, joker_mult, joker_money, mult_multiplier, messages) =
569                self.process_joker_effects(&hand);
570            self.chips += joker_chips as f64;
571            self.mult += joker_mult as f64;
572            self.money += joker_money as f64;
573
574            // Trigger scaling events for money gained
575            if joker_money > 0 {
576                for _ in 0..joker_money {
577                    self.process_scaling_event(ScalingEvent::MoneyGained);
578                }
579            }
580
581            // Apply mult multiplier to the final mult value
582            if mult_multiplier != 1.0 {
583                self.mult *= mult_multiplier;
584            }
585
586            // Log debug messages if enabled
587            for message in messages {
588                self.add_debug_message(message);
589            }
590        }
591
592        // compute score
593        let score = self.chips * self.mult;
594
595        // Check for killscreen condition
596        if !score.is_finite() {
597            self.add_debug_message("KILLSCREEN: Final score reached infinity!".to_string());
598        }
599
600        // reset chips and mult
601        self.mult = self.config.base_mult as f64;
602        self.chips = self.config.base_chips as f64;
603        score
604    }

606    /// Process JokerEffect from all jokers and return accumulated effects
607    pub fn process_joker_effects(&mut self, hand: &MadeHand) -> (i32, i32, i32, f64, Vec<String>) {
608        use crate::hand::Hand;
609
610        let mut messages = Vec::new();
611
612        // Create game context
613        let mut context = GameContext {
614            chips: self.chips as i32,
615            mult: self.mult as i32,
616            money: self.money as i32,
617            ante: self.ante_current as u8,
618            round: self.round as u32,
619            stage: &self.stage,
620            hands_played: 0,  // TODO: track this properly
621            discards_used: 0, // TODO: track this properly
622            jokers: &self.jokers,
623            hand: &Hand::new(hand.hand.cards().to_vec()),
624            discarded: &self.discarded,
625            joker_state_manager: &self.joker_state_manager,
626            hand_type_counts: &self.hand_type_counts,
627            cards_in_deck: self.deck.len(),
628            stone_cards_in_deck: 0, // TODO: Track stone cards when implemented
629            rng: &self.rng,
630        };
631
632        // Process hand-level effects using the cached processor
633        let select_hand = SelectHand::new(hand.hand.cards().to_vec());
634        let hand_result = self.joker_effect_processor.process_hand_effects(
635            &self.jokers,
636            &mut context,
637            &select_hand,
638        );
639
640        let mut total_chips = hand_result.accumulated_effect.chips;
641        let mut total_mult = hand_result.accumulated_effect.mult;
642        let mut total_money = hand_result.accumulated_effect.money;
643        let mut total_mult_multiplier = if hand_result.accumulated_effect.mult_multiplier != 0.0 {
644            hand_result.accumulated_effect.mult_multiplier
645        } else {
646            1.0
647        };
648
649        // Generate debug messages for hand effects
650        #[cfg(debug_assertions)]
651        if hand_result.jokers_processed > 0
652            && (total_chips != 0 || total_mult != 0 || total_money != 0)
653        {
654            use std::fmt::Write;
655            let mut debug_msg = String::with_capacity(128);
656            write!(
657                &mut debug_msg,
658                "Hand effects: +{} chips, +{} mult, +{} money from {} jokers",
659                total_chips, total_mult, total_money, hand_result.jokers_processed
660            )
661            .unwrap();
662
663            if hand_result.retriggered_count > 0 {
664                write!(
665                    &mut debug_msg,
666                    " ({} retriggers)",
667                    hand_result.retriggered_count
668                )
669                .unwrap();
670            }
671
672            messages.push(debug_msg);
673        }
674
675        // Process any error messages from hand effects
676        for error in &hand_result.errors {
677            if let crate::joker_effect_processor::EffectProcessingError::TooManyRetriggers(_) =
678                error
679            {
680                messages.push("KILLSCREEN: Too many retriggered effects!".to_string());
681            } // Other errors are less critical for gameplay
682        }
683
684        // Process card-level effects using the cached processor
685        for card in hand.hand.cards() {
686            let card_result =
687                self.joker_effect_processor
688                    .process_card_effects(&self.jokers, &mut context, &card);
689            // Accumulate card effects
690            total_chips += card_result.accumulated_effect.chips;
691            total_mult += card_result.accumulated_effect.mult;
692            total_money += card_result.accumulated_effect.money;
693
694            // Handle mult_multiplier: only apply if it's not the default value
695            if card_result.accumulated_effect.mult_multiplier != 0.0 {
696                total_mult_multiplier *= card_result.accumulated_effect.mult_multiplier;
697
698                // Killscreen detection
699                if !total_mult_multiplier.is_finite() {
700                    messages.push("KILLSCREEN: Score calculation reached infinity!".to_string());
701                    break;
702                }
703            }
704
705            // Generate debug messages for card effects
706            #[cfg(debug_assertions)]
707            if card_result.jokers_processed > 0
708                && (card_result.accumulated_effect.chips != 0
709                    || card_result.accumulated_effect.mult != 0
710                    || card_result.accumulated_effect.money != 0)
711            {
712                use std::fmt::Write;
713                let mut debug_msg = String::with_capacity(128);
714                write!(
715                    &mut debug_msg,
716                    "Card {} effects: +{} chips, +{} mult, +{} money",
717                    card,
718                    card_result.accumulated_effect.chips,
719                    card_result.accumulated_effect.mult,
720                    card_result.accumulated_effect.money
721                )
722                .unwrap();
723
724                if card_result.retriggered_count > 0 {
725                    write!(
726                        &mut debug_msg,
727                        " ({} retriggers)",
728                        card_result.retriggered_count
729                    )
730                    .unwrap();
731                }
732                messages.push(debug_msg);
733            }
734
735            // Process any error messages from card effects
736            for error in &card_result.errors {
737                if let crate::joker_effect_processor::EffectProcessingError::TooManyRetriggers(_) =
738                    error
739                {
740                    messages.push("KILLSCREEN: Too many retriggered effects!".to_string());
741                } // Other errors are less critical for gameplay
742            }
743        }
744
745        // Don't apply mult multiplier here - let calc_score handle it
746        // if total_mult_multiplier != 1.0 {
747        //     total_mult = (total_mult as f64 * total_mult_multiplier) as i32;
748        // }
749
750        (
751            total_chips,
752            total_mult,
753            total_money,
754            total_mult_multiplier,
755            messages,
756        )
757    }

760    pub fn calc_score_with_breakdown(&mut self, hand: MadeHand) -> ScoreBreakdown {
761        use crate::hand::Hand;
762
763        // Track initial values (for potential future use)
764        let _initial_chips = self.chips;
765        let _initial_mult = self.mult;
766

769        let base_mult = hand.rank.level().mult as f64;
770        self.chips += base_chips;
771        self.mult += base_mult;
772

774        let card_chips: f64 = hand.hand.cards().iter().map(|c| c.chips() as f64).sum();
775        self.chips += card_chips;

779        // Process jokers if any exist
780        if !self.jokers.is_empty() {

784                mult: self.mult as i32,
785                money: self.money as i32,
786                ante: self.ante_current as u8,
787                round: self.round as u32,
788                stage: &self.stage,
789                hands_played: 0,  // TODO: track this properly
790                discards_used: 0, // TODO: track this properly
791                jokers: &self.jokers,
792                hand: &Hand::new(hand.hand.cards().to_vec()),
793                discarded: &self.discarded,
794                joker_state_manager: &self.joker_state_manager,
795                hand_type_counts: &self.hand_type_counts,
796                cards_in_deck: self.deck.len(),
797                stone_cards_in_deck: 0, // TODO: Track stone cards when implemented
798                rng: &self.rng,
799            };
800
801            // Process each joker and track contributions
802            for joker in &self.jokers {
803                let select_hand = SelectHand::new(hand.hand.cards().to_vec());
804                let effect = joker.on_hand_played(&mut context, &select_hand);
805
806                // Create contribution record
807                let contribution = JokerContribution {
808                    joker_name: joker.name().to_string(),
809                    joker_id: joker.id(),
810                    chips_added: effect.chips,
811                    mult_added: effect.mult,
812                    mult_multiplier: if effect.mult_multiplier != 0.0 {
813                        effect.mult_multiplier
814                    } else {
815                        1.0
816                    },
817                    money_added: effect.money,
818                    retrigger_count: effect.retrigger,
819                };
820
821                // Apply effects to game state
822                self.chips += effect.chips as f64;
823                self.mult += effect.mult as f64;
824                self.money += effect.money as f64;
825
826                joker_contributions.push(contribution);
827            }
828
829            // Process card-level effects for each joker
830            for card in hand.hand.cards() {
831                for joker in &self.jokers {
832                    let effect = joker.on_card_scored(&mut context, &card);
833
834                    if effect.chips != 0 || effect.mult != 0 || effect.money != 0 {
835                        // Find existing contribution for this joker or create new one
836                        if let Some(existing) = joker_contributions
837                            .iter_mut()
838                            .find(|c| c.joker_id == joker.id())
839                        {
840                            existing.chips_added += effect.chips;
841                            existing.mult_added += effect.mult;
842                            existing.money_added += effect.money;
843                        }
844
845                        // Apply effects to game state
846                        self.chips += effect.chips as f64;
847                        self.mult += effect.mult as f64;
848                        self.money += effect.money as f64;
849                    }
850                }
851            }
852
853            // Trigger scaling events for hand played
854            let hand_rank = hand.rank;
855            self.process_scaling_event(ScalingEvent::HandPlayed(hand_rank));
856        }
857
858        // Calculate final score
859        let final_score = self.chips * self.mult;
860
861        // Log final breakdown if debug enabled (optimized)
862        #[cfg(debug_assertions)]
863        {
864            use std::fmt::Write;
865            let mut msg = String::with_capacity(64);
866            write!(
867                &mut msg,
868                "Final score: {} chips × {} mult = {}",
869                self.chips, self.mult, final_score
870            )
871            .unwrap();
872            self.add_debug_message(msg);
873        }
874
875        // Reset chips and mult to base values
876        self.mult = self.config.base_mult as f64;
877        self.chips = self.config.base_chips as f64;
878
879        ScoreBreakdown {
880            base_chips,
881            base_mult,
882            card_chips,
883            joker_contributions,
884            final_score,
885        }
886    }

1416    fn select_blind(&mut self, blind: Blind) -> Result<(), GameError> {
1417        // can only set blind if stage is pre blind
1418        if self.stage != Stage::PreBlind() {
1419            return Err(GameError::InvalidStage);
1420        }
1421        // provided blind must be expected next blind
1422        if let Some(current) = self.blind {
1423            if blind != current.next() {
1424                return Err(GameError::InvalidBlind);
1425            }
1426        } else {
1427            // if game just started, blind will be None, in which case
1428            // we can only set it to small.
1429            if blind != Blind::Small {
1430                return Err(GameError::InvalidBlind);
1431            }
1432        }
1433        self.blind = Some(blind);
1434        self.stage = Stage::Blind(blind);
1435        self.deal();
1436        Ok(())
1437    }

1445    // Returns true if should clear blind after, false if not.
1446    fn handle_score(&mut self, score: f64) -> Result<bool, GameError> {
1447        // can only handle score if stage is blind
1448        if !self.stage.is_blind() {
1449            return Err(GameError::InvalidStage);
1450        }
1451
1452        self.score += score;
1453        let required = self.required_score();
1454
1455        // blind not passed
1456        if self.score < required {
1457            // no more hands to play -> lose
1458            if self.plays == 0.0 {
1459                self.stage = Stage::End(End::Lose);
1460                return Ok(false);
1461            } else {
1462                // more hands to play, carry on
1463                return Ok(false);
1464            }
1465        }
1466
1467        let blind = self.blind.ok_or(GameError::MissingBlindState)?;
1468        // score exceeds blind (blind passed).
1469        // handle reward then progress to next stage.
1470        let reward = self.calc_reward(blind)?;
1471        self.reward = reward;
1472
1473        // passed boss blind, either win or progress ante
1474        if blind == Blind::Boss {
1475            if let Some(ante_next) = self.ante_current.next(self.ante_end) {
1476                self.ante_current = ante_next;
1477            } else {
1478                self.stage = Stage::End(End::Win);
1479                return Ok(false);
1480            }
1481        };
1482
1483        // finish blind, proceed to post blind
1484        self.stage = Stage::PostBlind();
1485        Ok(true)
1486    }
