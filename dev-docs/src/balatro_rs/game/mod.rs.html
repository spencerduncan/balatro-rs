92    /// Accumulate a JokerEffect into this accumulator
93    /// Returns false if killscreen (infinity/NaN) is detected
94    pub fn accumulate_effect(&mut self, effect: &crate::joker::JokerEffect) -> bool {
95        // Calculate total iterations (1 + retriggers)
96        let iterations = 1 + effect.retrigger as i32;
97
98        // Accumulate chips, mult, and money (additive, multiplied by retriggers)
99        self.chips = self
100            .chips
101            .saturating_add(effect.chips.saturating_mul(iterations));
102        self.mult = self
103            .mult
104            .saturating_add(effect.mult.saturating_mul(iterations));
105        self.money = self
106            .money
107            .saturating_add(effect.money.saturating_mul(iterations));
108        self.interest_bonus = self
109            .interest_bonus
110            .saturating_add(effect.interest_bonus.saturating_mul(iterations));
111
112        // Apply mult multiplier for each iteration (multiplicative)
113        if effect.mult_multiplier != 0.0 {
114            for _ in 0..iterations {
115                self.mult_multiplier *= effect.mult_multiplier;
116            }
117        }
118
119        // Add message if present
120        if let Some(ref message) = effect.message {
121            self.messages.push(message.clone());
122        }
123
124        // Check for killscreen (infinity or NaN)
125        if self.mult_multiplier.is_infinite() || self.mult_multiplier.is_nan() {
126            return false;
127        }
128
129        true
130    }

132    /// Merge another AccumulatedEffects into this one
133    pub fn merge(&mut self, other: &AccumulatedEffects) {
134        self.chips = self.chips.saturating_add(other.chips);
135        self.mult = self.mult.saturating_add(other.mult);
136        self.money = self.money.saturating_add(other.money);
137
138        // Multiply mult multipliers
139        if other.mult_multiplier != 0.0 {
140            self.mult_multiplier *= other.mult_multiplier;
141        }
142
143        // Append messages
144        self.messages.extend_from_slice(&other.messages);
145    }

147    /// Apply the accumulated mult multiplier to the mult value
148    pub fn apply_mult_multiplier(&mut self) {
149        if self.mult_multiplier != 0.0 && self.mult_multiplier != 1.0 {
150            let multiplied = (self.mult as f64) * self.mult_multiplier;
151            // Clamp to i32 range to prevent overflow
152            self.mult = multiplied.clamp(i32::MIN as f64, i32::MAX as f64) as i32;
153        }
154    }

276/// Format debug message for joker effects with conditional compilation
277#[cfg(debug_assertions)]
278fn _format_joker_effect_debug_message(
279    joker_name: &str,
280    effect: &crate::joker::JokerEffect,
281    total_triggers: u32,
282    card: Option<&Card>,
283) -> String {
284    use std::fmt::Write;
285    let mut debug_msg = String::with_capacity(128); // Pre-allocate reasonable size
286
287    match card {
288        None => {
289            // Hand-level format
290            write!(
291                &mut debug_msg,
292                "Joker '{}': +{} chips, +{} mult, +{} money",
293                joker_name,
294                effect.chips * total_triggers as i32,
295                effect.mult * total_triggers as i32,
296                effect.money * total_triggers as i32
297            )
298            .unwrap();
299        }
300        Some(card) => {
301            // Card-level format
302            write!(
303                &mut debug_msg,
304                "Joker '{}' on card {}: +{} chips, +{} mult, +{} money",
305                joker_name,
306                card,
307                effect.chips * total_triggers as i32,
308                effect.mult * total_triggers as i32,
309                effect.money * total_triggers as i32
310            )
311            .unwrap();
312        }
313    }
314
315    if effect.retrigger > 0 {
316        write!(&mut debug_msg, " (retrigger x{})", effect.retrigger).unwrap();
317    }
318
319    debug_msg

642    pub fn calc_score(&mut self, hand: MadeHand) -> f64 {
643        // compute chips and mult from hand level
644        self.chips += hand.rank.level().chips as f64;
645        self.mult += hand.rank.level().mult as f64;
646
647        // add chips for each played card
648        let card_chips: f64 = hand.hand.cards().iter().map(|c| c.chips() as f64).sum();
649        self.chips += card_chips;
650
651        // Apply JokerEffect from structured joker system
652        if !self.jokers.is_empty() {
653            let (joker_chips, joker_mult, joker_money, mult_multiplier, messages) =
654                self.process_joker_effects(&hand);
655            self.chips += joker_chips as f64;
656            self.mult += joker_mult as f64;
657            self.money += joker_money as f64;
658
659            // Trigger scaling events for money gained
660            if joker_money > 0 {
661                for _ in 0..joker_money {
662                    self.process_scaling_event(ScalingEvent::MoneyGained);
663                }
664            }
665
666            // Apply mult multiplier to the final mult value
667            if mult_multiplier != 1.0 {
668                self.mult *= mult_multiplier;
669            }
670
671            // Log debug messages if enabled
672            for message in messages {
673                self.add_debug_message(message);
674            }
675        }
676
677        // compute score
678        let score = self.chips * self.mult;
679
680        // Check for killscreen condition
681        if !score.is_finite() {
682            self.add_debug_message("KILLSCREEN: Final score reached infinity!".to_string());
683        }
684
685        // reset chips and mult
686        self.mult = self.config.base_mult as f64;
687        self.chips = self.config.base_chips as f64;
688        score
689    }

898                    joker_id: joker.id(),

907                    retrigger_count: effect.retrigger,

912                self.mult += effect.mult as f64;
913                self.money += effect.money as f64;

936                        self.mult += effect.mult as f64;

938                    }
939                }
940            }

944            self.process_scaling_event(ScalingEvent::HandPlayed(hand_rank));
945        }
946

948        let final_score = self.chips * self.mult;

960            .unwrap();
961            self.add_debug_message(msg);
962        }
963
964        // Reset chips and mult to base values
965        self.mult = self.config.base_mult as f64;
966        self.chips = self.config.base_chips as f64;
967
968        ScoreBreakdown {
969            base_chips,
970            base_mult,
971            card_chips,
972            joker_contributions,
973            final_score,
974        }
975    }

1284    /// Purchases a joker from the shop and places it at the specified slot.
1285    ///
1286    /// This method validates that the game is in the shop stage, the joker is available
1287    /// in the shop, the player has sufficient money, and the slot is valid. If all
1288    /// validations pass, it purchases the joker and inserts it at the specified position,
1289    /// shifting existing jokers to the right if necessary.
1290    ///
1291    /// # Arguments
1292    /// * `joker_id` - The identifier of the joker to purchase
1293    /// * `slot` - The zero-based index where to place the joker (0 to jokers.len())
1294    ///
1295    /// # Returns
1296    /// * `Ok(())` if the purchase was successful
1297    /// * `Err(GameError)` if the purchase failed due to validation errors
1298    ///
1299    /// # Errors
1300    /// * `InvalidStage` - Game is not in shop stage
1301    /// * `InvalidSlot` - Slot index is greater than current joker count
1302    /// * `NoAvailableSlot` - Joker limit reached and trying to add at the end
1303    /// * `JokerNotInShop` - Requested joker is not available in the shop
1304    /// * `InvalidBalance` - Player doesn't have enough money
1305    /// * `NoJokerMatch` - Joker found in shop but couldn't be matched (internal error)
1306    pub(crate) fn buy_joker_with_slot(
1307        &mut self,
1308        joker_id: JokerId,
1309        slot: usize,
1310    ) -> Result<(), GameError> {
1311        // Validate stage
1312        if self.stage != Stage::Shop() {
1313            return Err(GameError::InvalidStage);
1314        }
1315
1316        // Validate slot index - must be within expanded joker slot limit
1317        if slot >= self.config.joker_slots {
1318            return Err(GameError::InvalidSlot);
1319        }
1320
1321        // Check if we've reached the joker limit
1322        if self.jokers.len() >= self.config.joker_slots {
1323            return Err(GameError::NoAvailableSlot);
1324        }
1325

1742    // Returns true if should clear blind after, false if not.
1743    fn handle_score(&mut self, score: f64) -> Result<bool, GameError> {
1744        // can only handle score if stage is blind
1745        if !self.stage.is_blind() {
1746            return Err(GameError::InvalidStage);
1747        }
1748
1749        self.score += score;
1750        let required = self.required_score();
1751
1752        // blind not passed
1753        if self.score < required {
1754            // no more hands to play -> lose
1755            if self.plays == 0.0 {
1756                self.stage = Stage::End(End::Lose);
1757                return Ok(false);
1758            } else {
1759                // more hands to play, carry on
1760                return Ok(false);
1761            }
1762        }
1763
1764        let blind = self.blind.ok_or(GameError::MissingBlindState)?;
1765        // score exceeds blind (blind passed).
1766        // process joker round end effects first
1767        let joker_effects = self.process_joker_round_end_effects();
1768
1769        // handle reward calculation with joker interest bonus
1770        let reward = self.calc_reward_with_interest_bonus(blind, joker_effects.interest_bonus)?;
1771        self.reward = reward;
1772
1773        // apply joker money effects (separate from interest)
1774        self.money += joker_effects.money as f64;
1775
1776        // passed boss blind, either win or progress ante
1777        if blind == Blind::Boss {
1778            if let Some(ante_next) = self.ante_current.next(self.ante_end) {
1779                self.ante_current = ante_next;
1780            } else {
1781                self.stage = Stage::End(End::Win);
1782                return Ok(false);
1783            }
1784        };
1785
1786        // finish blind, proceed to post blind
1787        self.stage = Stage::PostBlind();
1788        Ok(true)
1789    }

1952    /// Handle skipping a blind and potentially getting skip tags
1953    fn handle_skip_blind(&mut self, blind: Blind) -> Result<(), GameError> {
1954        use crate::skip_tags::tag_registry::global_registry;
1955        use crate::skip_tags::SkipTagInstance;
1956
1957        // First, call the original skip_blind method to set up the basic skip mechanics
1958        self.skip_blind(blind)?;
1959
1960        // Generate potential skip tags based on rarity weights
1961        let registry = global_registry();
1962        let weighted_tags = registry.get_weighted_tags();
1963
1964        if !weighted_tags.is_empty() {
1965            // For utility tags, generate one tag with some probability
1966            let tag_chance = 0.5; // 50% chance to get a tag when skipping
1967
1968            if self.rng.gen_range(0.0..1.0) < tag_chance {
1969                // Select a weighted random tag
1970                let total_weight: f64 = weighted_tags.iter().map(|(_, weight)| weight).sum();
1971                let mut random_value = self.rng.gen_range(0.0..total_weight);
1972
