173// Each card gets a unique id. Not sure this is strictly
174// necessary but it makes identifying otherwise identical cards
175// possible (i.e. for trashing, reordering, etc)
176static CARD_ID_COUNTER: AtomicUsize = AtomicUsize::new(0);
177
178#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
179#[cfg_attr(feature = "python", pyclass)]
180#[derive(PartialEq, PartialOrd, Eq, Ord, Clone, Copy, Hash)]
181pub struct Card {
182    pub value: Value,
183    pub suit: Suit,
184    pub id: usize,
185    pub edition: Edition,
186    pub enhancement: Option<Enhancement>,
187    pub seal: Option<Seal>,
188}
189
190impl Card {
191    pub fn new(value: Value, suit: Suit) -> Self {
192        let id = CARD_ID_COUNTER.fetch_add(1, Ordering::SeqCst);
193        Self {
194            value,
195            suit,
196            id,
197            edition: Edition::Base,
198            enhancement: None,
199            seal: None,
200        }
201    }
202
203    pub fn is_face(&self) -> bool {
204        matches!(self.value, Value::Jack | Value::Queen | Value::King)
205    }
206
207    pub fn is_even(&self) -> bool {
208        self.value != Value::Ace && !self.is_face() && self.value as u16 % 2 == 0
209    }
210
211    pub fn is_odd(&self) -> bool {
212        self.value == Value::Ace || !self.is_face() && self.value as u16 % 2 != 0
213    }
214
215    pub fn chips(&self) -> usize {
216        let base_chips = match self.value {
217            Value::Two => 1,
218            Value::Three => 2,
219            Value::Four => 3,
220            Value::Five => 4,
221            Value::Six => 5,
222            Value::Seven => 6,
223            Value::Eight => 7,
224            Value::Nine => 8,
225            Value::Ten => 9,
226            Value::Jack => 10,
227            Value::Queen => 10,
228            Value::King => 10,
229            Value::Ace => 11,
230        };
231
232        // Add edition bonuses - Foil edition gives +50 chips
233        match self.edition {
234            Edition::Foil => base_chips + 50,
235            _ => base_chips,
236        }
237    }
238}

240impl fmt::Debug for Card {
241    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
242        #[cfg(feature = "colored")]
243        let suit = match self.suit {
244            Suit::Spade => self.suit.unicode().bold(),
245            Suit::Club => self.suit.unicode().green().bold(),
246            Suit::Heart => self.suit.unicode().red().bold(),
247            Suit::Diamond => self.suit.unicode().blue().bold(),
248        };
249        #[cfg(not(feature = "colored"))]
250        let suit = self.suit.unicode();
251        write!(f, "Card({}{})", char::from(self.value), suit)
252    }
253}

317    #[test]
318    fn test_edition_bonus_foil_chips() {
319        // Test Foil edition gives +50 chips bonus
320        let mut card = Card::new(Value::Ace, Suit::Heart);
321        card.edition = Edition::Foil;
322
323        // Ace normally gives 11 chips, Foil should add 50 for total of 61
324        assert_eq!(card.chips(), 61);
325
326        // Test with different card values
327        let mut two = Card::new(Value::Two, Suit::Spade);
328        two.edition = Edition::Foil;
329        assert_eq!(two.chips(), 51); // 1 + 50
330
331        let mut king = Card::new(Value::King, Suit::Diamond);
332        king.edition = Edition::Foil;
333        assert_eq!(king.chips(), 60); // 10 + 50
334    }

364    #[test]
365    fn test_edition_bonus_production_edge_cases() {
366        // Production-ready edge case testing
367
368        // Test with all card values for Foil edition
369        for value in &Value::values() {
370            let mut card = Card::new(*value, Suit::Heart);
371            card.edition = Edition::Foil;
372
373            let base_chips = Card::new(*value, Suit::Heart).chips();
374            assert_eq!(
375                card.chips(),
376                base_chips + 50,
377                "Foil edition should add exactly 50 chips to {value:?}"

379        }
380
381        // Test overflow safety (should not panic even with extreme values)
382        let mut max_card = Card::new(Value::Ace, Suit::Heart); // Highest base chips (11)
383        max_card.edition = Edition::Foil;
384        let result = max_card.chips();
385        assert_eq!(result, 61);
386        assert!(result < usize::MAX, "Should not overflow");
