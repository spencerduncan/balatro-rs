5use serde::{Deserialize, Serialize};

7/// Default maximum number of actions to keep in history
8pub const DEFAULT_MAX_ACTIONS: usize = 10_000;
9
10/// Bounded circular buffer for action history to prevent memory leaks
11///
12/// This replaces the unbounded `Vec<Action>` with a memory-efficient circular buffer
13/// that maintains only the most recent actions up to a configurable limit.
14#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
15#[derive(Debug, Clone)]
16pub struct BoundedActionHistory {
17    /// Internal circular buffer storing actions
18    actions: VecDeque<Action>,
19    /// Maximum number of actions to store
20    max_size: usize,
21    /// Total number of actions that have been recorded (including evicted ones)
22    total_actions: usize,
23}
24
25impl BoundedActionHistory {
26    /// Create a new bounded action history with the default size limit
27    pub fn new() -> Self {
28        Self::with_capacity(DEFAULT_MAX_ACTIONS)
29    }
30
31    /// Create a new bounded action history with a specific size limit
32    pub fn with_capacity(max_size: usize) -> Self {
33        Self {
34            actions: VecDeque::with_capacity(max_size.min(DEFAULT_MAX_ACTIONS)),
35            max_size,
36            total_actions: 0,
37        }
38    }
39
40    /// Add a new action to the history
41    ///
42    /// If the history is at capacity, the oldest action will be evicted
43    pub fn push(&mut self, action: Action) {
44        if self.actions.len() >= self.max_size {
45            self.actions.pop_front();
46        }
47        self.actions.push_back(action);
48        self.total_actions += 1;
49    }
50
51    /// Get all actions in the history as a Vec
52    ///
53    /// This returns a clone of the current actions for compatibility with existing code
54    /// Note: This method is expensive and should be used sparingly
55    pub fn to_vec(&self) -> Vec<Action> {
56        self.actions.iter().cloned().collect()
57    }
58
59    /// Get the number of actions currently stored
60    pub fn len(&self) -> usize {
61        self.actions.len()
62    }
63
64    /// Check if the history is empty
65    pub fn is_empty(&self) -> bool {
66        self.actions.is_empty()
67    }
68
69    /// Get the total number of actions that have been recorded (including evicted ones)
70    pub fn total_actions(&self) -> usize {
71        self.total_actions
72    }
73
74    /// Get the maximum capacity of the history
75    pub fn capacity(&self) -> usize {
76        self.max_size
77    }
78
79    /// Clear all actions from the history
80    pub fn clear(&mut self) {
81        self.actions.clear();
82        self.total_actions = 0;
83    }
84
85    /// Get an iterator over the actions (most recent first)
86    pub fn iter_recent_first(&self) -> impl Iterator<Item = &Action> {
87        self.actions.iter().rev()
88    }
89
90    /// Get an iterator over the actions (oldest first)  
91    pub fn iter(&self) -> impl Iterator<Item = &Action> {
92        self.actions.iter()
93    }
94
95    /// Get the most recent action, if any
96    pub fn last(&self) -> Option<&Action> {
97        self.actions.back()
98    }
99
100    /// Get the oldest action in the current history, if any
101    pub fn first(&self) -> Option<&Action> {
102        self.actions.front()
103    }
104
105    /// Resize the history capacity
106    ///
107    /// If the new capacity is smaller than the current number of actions,
108    /// the oldest actions will be evicted
109    pub fn resize(&mut self, new_max_size: usize) {
110        self.max_size = new_max_size;
111        while self.actions.len() > self.max_size {
112            self.actions.pop_front();
113        }
114        // Also shrink the underlying capacity if significantly larger
115        if self.actions.capacity() > self.max_size * 2 {
116            self.actions.shrink_to_fit();
117        }
118    }
119
120    /// Get memory usage statistics
121    pub fn memory_stats(&self) -> MemoryStats {
122        MemoryStats {
123            current_actions: self.len(),
124            max_capacity: self.capacity(),
125            total_recorded: self.total_actions(),
126            estimated_bytes: self.len() * std::mem::size_of::<Action>()
127                + std::mem::size_of::<Self>(),
128        }
129    }
130}

132/// Memory usage statistics for bounded action history
133#[derive(Debug, Clone)]
134pub struct MemoryStats {
135    pub current_actions: usize,
136    pub max_capacity: usize,
137    pub total_recorded: usize,
138    pub estimated_bytes: usize,
139}

141impl Default for BoundedActionHistory {
142    fn default() -> Self {
143        Self::new()
144    }
145}
146
147impl From<Vec<Action>> for BoundedActionHistory {
148    fn from(actions: Vec<Action>) -> Self {
149        let total_len = actions.len();
150        let mut history = Self::with_capacity(DEFAULT_MAX_ACTIONS);
151
152        // If we have more actions than capacity, only keep the most recent ones
153        let start_idx = total_len.saturating_sub(DEFAULT_MAX_ACTIONS);
154
155        for action in actions.into_iter().skip(start_idx) {
156            history.actions.push_back(action);
157        }
158
159        history.total_actions = total_len;
160        history
161    }
162}

164#[cfg(test)]
165mod tests {

169    #[test]
170    fn test_bounded_action_history_basic() {
171        let mut history = BoundedActionHistory::with_capacity(3);
172
173        assert_eq!(history.len(), 0);
174        assert!(history.is_empty());
175
176        history.push(Action::Play());
177        history.push(Action::Discard());
178        assert_eq!(history.len(), 2);
179        assert_eq!(history.total_actions(), 2);
180    }
181
182    #[test]
183    fn test_bounded_action_history_overflow() {
184        let mut history = BoundedActionHistory::with_capacity(2);
185
186        history.push(Action::Play());
187        history.push(Action::Discard());
188        history.push(Action::NextRound());
189
190        // Should only have 2 actions (most recent)
191        assert_eq!(history.len(), 2);
192        assert_eq!(history.total_actions(), 3);
193
194        let actions = history.to_vec();
195        assert_eq!(actions, vec![Action::Discard(), Action::NextRound()]);
196    }
197
198    #[test]
199    fn test_clear() {
200        let mut history = BoundedActionHistory::with_capacity(5);
201        history.push(Action::Play());
202        history.push(Action::Discard());
203
204        history.clear();
205        assert_eq!(history.len(), 0);
206        assert_eq!(history.total_actions(), 0);
207    }

209    #[test]

212        for _ in 0..5 {
