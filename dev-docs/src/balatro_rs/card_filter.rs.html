11/// Context information passed to card filters
12/// Provides access to game state and other contextual information needed for filtering
13#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
14#[derive(Debug, Clone)]
15pub struct FilterContext {
16    /// Current game state (optional - not all filters need game context)
17    pub game_state: Option<GameStateSnapshot>,
18    /// Additional context data for advanced filtering
19    pub metadata: FilterMetadata,
20}
21
22/// Snapshot of relevant game state for filtering
23#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
24#[derive(Debug, Clone)]
25pub struct GameStateSnapshot {
26    /// Current ante level
27    pub ante: u8,
28    /// Player's current money
29    pub money: i32,
30    /// Number of jokers owned
31    pub joker_count: usize,
32}
33
34/// Additional metadata for filtering context
35#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
36#[derive(Debug, Clone)]
37pub struct FilterMetadata {
38    /// Custom properties for extensibility
39    pub properties: std::collections::HashMap<String, String>,
40}
41
42impl FilterContext {
43    /// Create a new FilterContext with minimal data
44    pub fn new() -> Self {
45        Self {
46            game_state: None,
47            metadata: FilterMetadata {
48                properties: std::collections::HashMap::new(),
49            },
50        }
51    }
52
53    /// Create a FilterContext from game state
54    pub fn from_game(game: &Game) -> Self {
55        Self {
56            game_state: Some(GameStateSnapshot {
57                ante: match game.ante_current {
58                    crate::ante::Ante::Zero => 0,
59                    crate::ante::Ante::One => 1,
60                    crate::ante::Ante::Two => 2,
61                    crate::ante::Ante::Three => 3,
62                    crate::ante::Ante::Four => 4,
63                    crate::ante::Ante::Five => 5,
64                    crate::ante::Ante::Six => 6,
65                    crate::ante::Ante::Seven => 7,
66                    crate::ante::Ante::Eight => 8,
67                },
68                money: game.money as i32,
69                joker_count: game.jokers.len(),
70            }),
71            metadata: FilterMetadata {
72                properties: std::collections::HashMap::new(),
73            },
74        }
75    }
76
77    /// Add a metadata property
78    pub fn with_property(mut self, key: String, value: String) -> Self {
79        self.metadata.properties.insert(key, value);
80        self
81    }
82}

85    fn default() -> Self {
86        Self::new()

90/// Core trait for card filtering
91///
92/// All card filters must implement this trait. The `matches` method determines
93/// whether a given card satisfies the filter criteria.
94pub trait CardFilter {
95    /// Test if a card matches this filter's criteria
96    ///
97    /// # Arguments
98    /// * `card` - The card to test
99    /// * `context` - Optional context information for advanced filtering
100    ///
101    /// # Returns
102    /// `true` if the card matches the filter, `false` otherwise
103    fn matches(&self, card: &Card, context: &FilterContext) -> bool;
104
105    /// Get a human-readable description of this filter
106    fn description(&self) -> String {
107        "Card filter".to_string()
108    }
109}
110
111/// Enumeration of basic card filter types
112///
113/// These are the fundamental filter types that can be composed together
114/// for more complex filtering logic.
115#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
116#[derive(Debug, Clone, PartialEq, Eq)]
117pub enum CardFilterType {
118    /// Filter cards based on enhancement status
119    Enhanced,
120    /// Filter cards based on rarity (mapped to edition for now)
121    Rarity(Edition),
122    /// Filter cards based on suit
123    Suit(Suit),
124    /// Filter cards based on rank/value
125    Rank(Value),
126    /// Filter cards based on edition
127    Edition(Edition),
128    /// Filter cards based on seal presence
129    Sealed,
130    /// Filter face cards (Jack, Queen, King)
131    Face,
132    /// Filter even-valued cards
133    Even,
134    /// Filter odd-valued cards
135    Odd,
136}
137
138impl CardFilter for CardFilterType {
139    fn matches(&self, card: &Card, _context: &FilterContext) -> bool {
140        match self {
141            CardFilterType::Enhanced => card.enhancement.is_some(),
142            CardFilterType::Rarity(edition) => card.edition == *edition,
143            CardFilterType::Suit(suit) => card.suit == *suit,
144            CardFilterType::Rank(value) => card.value == *value,
145            CardFilterType::Edition(edition) => card.edition == *edition,
146            CardFilterType::Sealed => card.seal.is_some(),
147            CardFilterType::Face => card.is_face(),
148            CardFilterType::Even => card.is_even(),
149            CardFilterType::Odd => card.is_odd(),
150        }
151    }
152
153    fn description(&self) -> String {
154        match self {
155            CardFilterType::Enhanced => "Enhanced cards".to_string(),
156            CardFilterType::Rarity(edition) => format!("Cards with {edition:?} rarity"),
157            CardFilterType::Suit(suit) => format!("Cards of suit {suit:?}"),
158            CardFilterType::Rank(value) => format!("Cards with rank {value:?}"),
159            CardFilterType::Edition(edition) => format!("Cards with {edition:?} edition"),
160            CardFilterType::Sealed => "Cards with seals".to_string(),
161            CardFilterType::Face => "Face cards (J, Q, K)".to_string(),
162            CardFilterType::Even => "Even-valued cards".to_string(),
163            CardFilterType::Odd => "Odd-valued cards".to_string(),
164        }
165    }
166}

168/// Composite filter that combines multiple filters with logical operators
169#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
170#[derive(Debug, Clone)]
171pub enum CompositeFilter {
172    /// All filters must match (AND logic)
173    All(Vec<CardFilterType>),
174    /// Any filter must match (OR logic)
175    Any(Vec<CardFilterType>),
176    /// Filter must not match (NOT logic)
177    Not(Box<CardFilterType>),
178}
179
180impl CardFilter for CompositeFilter {
181    fn matches(&self, card: &Card, context: &FilterContext) -> bool {
182        match self {
183            CompositeFilter::All(filters) => {
184                filters.iter().all(|filter| filter.matches(card, context))
185            }
186            CompositeFilter::Any(filters) => {
187                filters.iter().any(|filter| filter.matches(card, context))
188            }
189            CompositeFilter::Not(filter) => !filter.matches(card, context),
190        }
191    }
192
193    fn description(&self) -> String {
194        match self {
195            CompositeFilter::All(filters) => {
196                let descriptions: Vec<String> = filters.iter().map(|f| f.description()).collect();
197                format!("All of: [{}]", descriptions.join(", "))
198            }
199            CompositeFilter::Any(filters) => {
200                let descriptions: Vec<String> = filters.iter().map(|f| f.description()).collect();
201                format!("Any of: [{}]", descriptions.join(", "))
202            }
203            CompositeFilter::Not(filter) => {
204                format!("Not: {}", filter.description())
205            }
206        }
207    }
208}

253/// Filter a collection of cards using the provided filter
254///
255/// # Arguments
256/// * `cards` - Iterator over cards to filter
257/// * `filter` - The filter to apply
258/// * `context` - Context information for filtering
259///
260/// # Returns
261/// A vector containing only the cards that match the filter
262pub fn filter_cards<'a, I, F>(cards: I, filter: &F, context: &FilterContext) -> Vec<&'a Card>
263where
264    I: Iterator<Item = &'a Card>,
265    F: CardFilter,
266{
267    cards.filter(|card| filter.matches(card, context)).collect()
268}

270/// Registry for managing and creating card filters
271///
272/// Provides a centralized way to create, register, and retrieve card filters.
273/// Supports both predefined filter types and custom filter definitions.
274#[derive(Debug, Clone)]
275pub struct CardFilterRegistry {
276    /// Named filters that can be referenced by string
277    named_filters: std::collections::HashMap<String, CardFilterType>,
278    /// Composite filters for complex logic
279    composite_filters: std::collections::HashMap<String, CompositeFilter>,
280}
281
282impl CardFilterRegistry {
283    /// Create a new empty registry
284    pub fn new() -> Self {
285        Self {
286            named_filters: std::collections::HashMap::new(),
287            composite_filters: std::collections::HashMap::new(),
288        }
289    }
290
291    /// Create a registry with default filters pre-registered
292    pub fn with_defaults() -> Self {
293        let mut registry = Self::new();
294
295        // Register common basic filters
296        registry.register("enhanced", CardFilterType::enhanced());
297        registry.register("sealed", CardFilterType::sealed());
298        registry.register("face", CardFilterType::face());
299        registry.register("even", CardFilterType::even());
300        registry.register("odd", CardFilterType::odd());
301
302        // Register suit filters
303        registry.register("hearts", CardFilterType::of_suit(Suit::Heart));
304        registry.register("diamonds", CardFilterType::of_suit(Suit::Diamond));
305        registry.register("clubs", CardFilterType::of_suit(Suit::Club));
306        registry.register("spades", CardFilterType::of_suit(Suit::Spade));
307
308        // Register edition filters
309        registry.register("foil", CardFilterType::of_edition(Edition::Foil));
310        registry.register(
311            "holographic",
312            CardFilterType::of_edition(Edition::Holographic),
313        );
314        registry.register(
315            "polychrome",
316            CardFilterType::of_edition(Edition::Polychrome),
317        );
318        registry.register("negative", CardFilterType::of_edition(Edition::Negative));
319
320        // Register some useful composite filters
321        registry.register_composite(
322            "red_cards",
323            CompositeFilter::Any(vec![
324                CardFilterType::of_suit(Suit::Heart),
325                CardFilterType::of_suit(Suit::Diamond),
326            ]),
327        );
328        registry.register_composite(
329            "black_cards",
330            CompositeFilter::Any(vec![
331                CardFilterType::of_suit(Suit::Club),
332                CardFilterType::of_suit(Suit::Spade),
333            ]),
334        );
335        registry.register_composite(
336            "special_edition",
337            CompositeFilter::Not(Box::new(CardFilterType::of_edition(Edition::Base))),
338        );
339
340        registry
341    }

343    /// Register a basic filter with a name
344    pub fn register(&mut self, name: &str, filter: CardFilterType) {
345        self.named_filters.insert(name.to_string(), filter);
346    }
347
348    /// Register a composite filter with a name
349    pub fn register_composite(&mut self, name: &str, filter: CompositeFilter) {
350        self.composite_filters.insert(name.to_string(), filter);
351    }
352
353    /// Get a basic filter by name
354    pub fn get_filter(&self, name: &str) -> Option<&CardFilterType> {
355        self.named_filters.get(name)
356    }
357
358    /// Get a composite filter by name
359    pub fn get_composite_filter(&self, name: &str) -> Option<&CompositeFilter> {
360        self.composite_filters.get(name)
361    }
362
363    /// Get all registered basic filter names
364    pub fn list_filters(&self) -> Vec<&String> {
365        self.named_filters.keys().collect()
366    }
367
368    /// Get all registered composite filter names
369    pub fn list_composite_filters(&self) -> Vec<&String> {
370        self.composite_filters.keys().collect()
371    }
372
373    /// Create a filter from a string definition
374    ///
375    /// Supports basic syntax like:
376    /// - "enhanced" - basic named filter
377    /// - "hearts" - suit filter
378    /// - "face AND enhanced" - composite filter (future extension)
379    pub fn create_from_string(&self, definition: &str) -> Result<Box<dyn CardFilter>, FilterError> {
380        let trimmed = definition.trim().to_lowercase();
381
382        // Try basic filter first
383        if let Some(filter) = self.get_filter(&trimmed) {
384            return Ok(Box::new(filter.clone()));
385        }
386
387        // Try composite filter
388        if let Some(filter) = self.get_composite_filter(&trimmed) {
389            return Ok(Box::new(filter.clone()));
390        }
391
392        // Could extend this to parse complex expressions in the future
393        Err(FilterError::UnknownFilter(definition.to_string()))
394    }
395
396    /// Apply a named filter to a collection of cards
397    pub fn apply_filter<'a>(
398        &self,
399        name: &str,
400        cards: &'a [Card],
401        context: &FilterContext,
402    ) -> Result<Vec<&'a Card>, FilterError> {
403        if let Some(filter) = self.get_filter(name) {
404            Ok(filter_cards(cards.iter(), filter, context))
405        } else if let Some(filter) = self.get_composite_filter(name) {
406            Ok(filter_cards(cards.iter(), filter, context))
407        } else {
408            Err(FilterError::UnknownFilter(name.to_string()))
409        }
410    }
411}

419/// Errors that can occur during filter operations
420#[derive(Debug, Clone, PartialEq)]
421pub enum FilterError {
422    /// Filter name was not found in the registry
423    UnknownFilter(String),
424    /// Invalid filter definition syntax
425    InvalidDefinition(String),
426}
427
428impl std::fmt::Display for FilterError {
429    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
430        match self {
431            FilterError::UnknownFilter(name) => write!(f, "Unknown filter: {name}"),
432            FilterError::InvalidDefinition(def) => write!(f, "Invalid filter definition: {def}"),
433        }
434    }
435}
