1//! # Error Handling Security Framework
2//!
3//! This module provides a security-focused error handling system that prevents
4//! information disclosure while maintaining developer productivity.
5//!
6//! ## Security Architecture
7//!
8//! The error system uses a dual-error approach:
9//! - **DeveloperError**: Detailed errors for internal debugging (never exposed to users)
10//! - **UserError**: Sanitized, generic errors safe for user consumption
11//!
12//! ## Error Detail Levels
13//!
14//! Three configuration levels control information disclosure:
15//! - `Production`: Only generic error messages (secure)
16//! - `Development`: Slightly more specific errors for debugging
17//! - `Testing`: Full error details for comprehensive testing
18//!
19//! ## Usage Guidelines
20//!
21//! ### For Library Developers
22//! ```rust
23//! use balatro_rs::error::{DeveloperGameError, UserError, ErrorSanitizer, ErrorDetailLevel};
24//!
25//! // Use DeveloperError types for internal error handling
26//! fn risky_operation() -> Result<(), DeveloperGameError> {
27//!     // Detailed error with sensitive information is OK here
28//!     Err(DeveloperGameError::JokerNotFound("internal_path_42".to_string()))
29//! }
30//!
31//! // Sanitize errors before exposing to users
32//! fn public_api() -> Result<(), UserError> {
33//!     let sanitizer = ErrorSanitizer::new(ErrorDetailLevel::Production);
34//!     match risky_operation() {
35//!         Ok(val) => Ok(val),
36//!         Err(dev_error) => Err(sanitizer.sanitize_game_error(&dev_error))
37//!     }
38//! }
39//! ```
40//!
41//! ### For Application Developers
42//! ```rust
43//! use balatro_rs::error::{UserError, ErrorDetailLevel, ErrorSanitizer};
44//!
45//! // Configure error detail level based on environment
46//! let detail_level = if cfg!(debug_assertions) {
47//!     ErrorDetailLevel::Development
48//! } else {
49//!     ErrorDetailLevel::Production
50//! };
51//!
52//! let sanitizer = ErrorSanitizer::new(detail_level);
53//! ```
54//!
55//! ## Security Best Practices
56//!
57//! 1. **Never expose DeveloperError directly to users**
58//! 2. **Always use ErrorSanitizer for user-facing errors**
59//! 3. **Log detailed errors internally for debugging**
60//! 4. **Use Production mode in release builds**
61//! 5. **Review error messages for information disclosure**
62//!
63//! ## Information Disclosure Prevention
64//!
65//! The system prevents disclosure of:
66//! - File paths and system internals
67//! - Stack traces and debug information
68//! - Database queries and connection details
69//! - User IDs and sensitive identifiers
70//! - Configuration values and secrets
71//!
72//! ## Backward Compatibility
73//!
74//! Type aliases maintain compatibility with existing code:
75//! ```rust
76//! use balatro_rs::error::{DeveloperGameError, DeveloperPlayHandError, DeveloperActionSpaceError};
77//!
78//! // These still work but are deprecated
79//! type GameError = DeveloperGameError;
80//! type PlayHandError = DeveloperPlayHandError;
81//! type ActionSpaceError = DeveloperActionSpaceError;
82//! ```

90/// Configuration for error detail levels to control information disclosure
91#[derive(Debug, Clone, Copy, PartialEq)]
92pub enum ErrorDetailLevel {
93    /// Production mode - only generic error messages (secure)
94    Production,
95    /// Development mode - detailed error messages for debugging
96    Development,
97    /// Testing mode - full error details for comprehensive testing
98    Testing,
99}
100
101impl Default for ErrorDetailLevel {
102    fn default() -> Self {
103        #[cfg(debug_assertions)]
104        return ErrorDetailLevel::Development;
105        #[cfg(not(debug_assertions))]
106        return ErrorDetailLevel::Production;
107    }
108}

315impl std::convert::From<DeveloperPackError> for DeveloperGameError {
316    fn from(err: DeveloperPackError) -> DeveloperGameError {
317        match err {
318            DeveloperPackError::InsufficientFunds { .. } => DeveloperGameError::InvalidBalance,
319            DeveloperPackError::PackNotFound { .. }
320            | DeveloperPackError::NoPackOpen
321            | DeveloperPackError::NoPackToSkip => DeveloperGameError::NoCardMatch,
322            DeveloperPackError::InvalidPackId { .. }
323            | DeveloperPackError::InvalidOptionIndex { .. }
324            | DeveloperPackError::PackIdMismatch { .. } => DeveloperGameError::InvalidAction,
325            DeveloperPackError::InvalidPurchaseStage { .. }
326            | DeveloperPackError::InvalidOpenStage { .. }
327            | DeveloperPackError::InvalidSelectionStage { .. } => DeveloperGameError::InvalidStage,
328            DeveloperPackError::RngSelectionFailed { .. }
329            | DeveloperPackError::WeightedSelectionFailed { .. } => {
330                DeveloperGameError::RngFailed("Pack generation RNG failure".to_string())
331            }
332            _ => DeveloperGameError::InvalidOperation(format!("Pack system error: {err}")),
333        }
334    }
335}

337/// Error sanitizer that converts detailed developer errors to safe user errors
338pub struct ErrorSanitizer {
339    detail_level: ErrorDetailLevel,
340}
341
342impl ErrorSanitizer {
343    pub fn new(detail_level: ErrorDetailLevel) -> Self {
344        Self { detail_level }
345    }
346
347    /// Convert developer errors to user-safe errors based on configuration
348    pub fn sanitize_game_error(&self, error: &DeveloperGameError) -> UserError {
349        match self.detail_level {
350            ErrorDetailLevel::Production => self.to_generic_user_error(error),
351            ErrorDetailLevel::Development | ErrorDetailLevel::Testing => {
352                // In development/testing, we can provide slightly more specific errors
353                // but still avoid exposing internal system details
354                self.to_specific_user_error(error)
355            }
356        }
357    }
358
359    fn to_generic_user_error(&self, error: &DeveloperGameError) -> UserError {
360        match error {
361            DeveloperGameError::InvalidInput(_) => UserError::InvalidInput,
362            DeveloperGameError::InvalidOperation(_) => UserError::InvalidOperation,
363            DeveloperGameError::NoCardMatch
364            | DeveloperGameError::NoJokerMatch
365            | DeveloperGameError::JokerNotFound(_)
366            | DeveloperGameError::JokerNotInShop => UserError::NotFound,
367            DeveloperGameError::InvalidHand(_)
368            | DeveloperGameError::InvalidStage
369            | DeveloperGameError::InvalidAction
370            | DeveloperGameError::InvalidBlind
371            | DeveloperGameError::InvalidBalance
372            | DeveloperGameError::InvalidActionSpace
373            | DeveloperGameError::InvalidSlot => UserError::InvalidState,
374            _ => UserError::SystemError,
375        }
376    }
377
378    fn to_specific_user_error(&self, error: &DeveloperGameError) -> UserError {
379        match error {
380            DeveloperGameError::InvalidInput(_) => UserError::InvalidInput,
381            DeveloperGameError::InvalidOperation(_) => UserError::InvalidOperation,
382            DeveloperGameError::NoCardMatch
383            | DeveloperGameError::NoJokerMatch
384            | DeveloperGameError::JokerNotFound(_)
385            | DeveloperGameError::JokerNotInShop => UserError::NotFound,
386            DeveloperGameError::InvalidHand(_)
387            | DeveloperGameError::InvalidStage
388            | DeveloperGameError::InvalidAction
389            | DeveloperGameError::InvalidBlind
390            | DeveloperGameError::InvalidBalance
391            | DeveloperGameError::InvalidActionSpace
392            | DeveloperGameError::InvalidSlot => UserError::InvalidState,
393            DeveloperGameError::NoRemainingDiscards | DeveloperGameError::NoRemainingPlays => {
394                UserError::OperationFailed
395            }
396            _ => UserError::SystemError,
397        }
398    }
399}
400
401impl Default for ErrorSanitizer {
402    fn default() -> Self {
403        Self::new(ErrorDetailLevel::default())
404    }
405}

412#[cfg(feature = "python")]
413impl std::convert::From<DeveloperGameError> for PyErr {
414    fn from(err: DeveloperGameError) -> PyErr {
415        // In Python bindings, always use sanitized errors for security
416        let sanitizer = ErrorSanitizer::default();
417        let user_error = sanitizer.sanitize_game_error(&err);
418        PyException::new_err(user_error.to_string())
419    }
420}
