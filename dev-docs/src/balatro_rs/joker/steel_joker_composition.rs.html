15/// Steel Joker - Gives X-Mult for each Steel Card in your full deck
16///
17/// According to joker.json: "Gives X#1# Mult for each Steel Card in your full deck"
18/// This implementation scans the deck composition using GameContext.steel_cards_in_deck
19/// instead of listening for destruction events.
20#[derive(Debug, Clone)]
21pub struct SteelJoker {
22    id: JokerId,
23    name: String,
24    description: String,
25    rarity: JokerRarity,
26    cost: usize,
27    /// X-Mult bonus per Steel Card (default: 0.25)
28    mult_per_steel_card: f64,
29}
30
31impl Default for SteelJoker {
32    fn default() -> Self {
33        Self::new()
34    }
35}
36
37impl SteelJoker {
38    pub fn new() -> Self {
39        Self::new_with_multiplier(0.25)
40    }
41
42    /// Create Steel Joker with custom multiplier per Steel Card
43    pub fn new_with_multiplier(mult_per_steel_card: f64) -> Self {
44        Self {
45            id: JokerId::SteelJoker,
46            name: "Steel Joker".to_string(),
47            description: format!(
48                "Gives X{mult_per_steel_card:.2} Mult for each Steel Card in your full deck"
49            ),
50            rarity: JokerRarity::Uncommon,
51            cost: 6,
52            mult_per_steel_card,
53        }
54    }
55
56    /// Calculate the current X-Mult multiplier based on Steel Cards in deck
57    fn calculate_multiplier(&self, steel_cards_count: usize) -> f64 {
58        1.0 + (self.mult_per_steel_card * steel_cards_count as f64)
59    }
60}

62impl JokerIdentity for SteelJoker {
63    fn joker_type(&self) -> &'static str {
64        "steel_joker"
65    }

69    }

72        &self.description

75    fn rarity(&self) -> Rarity {
76        match self.rarity {
77            JokerRarity::Common => Rarity::Common,
78            JokerRarity::Uncommon => Rarity::Uncommon,
79            JokerRarity::Rare => Rarity::Rare,
80            JokerRarity::Legendary => Rarity::Legendary,
81        }
82    }

85        self.cost as u64
86    }

89impl Joker for SteelJoker {

91        self.id
92    }

95        &self.name
96    }

99        &self.description

103        self.rarity
104    }
105
106    fn cost(&self) -> usize {
107        self.cost
108    }
109
110    /// Main joker logic: Scan deck composition for Steel Cards and apply X-Mult
111    fn on_hand_played(&self, context: &mut GameContext, _hand: &SelectHand) -> JokerEffect {
112        let steel_cards_count = context.steel_cards_in_deck;
113
114        if steel_cards_count == 0 {
115            // No Steel Cards in deck - return neutral multiplier (1.0 = no change)
116            return JokerEffect::new().with_mult_multiplier(1.0);
117        }
118
119        let current_multiplier = self.calculate_multiplier(steel_cards_count);
120
121        JokerEffect::new()
122            .with_mult_multiplier(current_multiplier)
123            .with_message(format!(
124            "Steel Joker: X{current_multiplier:.2} Mult ({steel_cards_count} Steel cards in deck)"
125        ))
126    }
127}

129impl JokerGameplay for SteelJoker {
130    fn process(&mut self, stage: &Stage, _context: &mut ProcessContext) -> ProcessResult {
131        if !matches!(stage, Stage::Blind(_)) {
132            return ProcessResult::default();
133        }
134
135        // Note: ProcessContext doesn't have access to game resources like steel_cards_in_deck
136        // The main logic is in the Joker trait's on_hand_played method which has GameContext
137        ProcessResult::default()
138    }
139
140    fn can_trigger(&self, stage: &Stage, _context: &ProcessContext) -> bool {
141        // Steel Joker can always trigger during blinds (deck composition check happens in on_hand_played)
142        matches!(stage, Stage::Blind(_))
143    }
144}
