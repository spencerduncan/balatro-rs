217/// Effect that a joker can apply to modify game state.
218///
219/// `JokerEffect` represents the comprehensive impact a joker can have on the game
220/// when triggered. Effects can modify scoring, award resources, trigger special
221/// mechanics, or even destroy jokers and transform cards.
222///
223/// # Effect Categories
224///
225/// ## Scoring Effects
226/// - **Chips**: Bonus chips added to hand total (`chips`)
227/// - **Mult**: Bonus mult added to hand total (`mult`)
228/// - **Mult Multiplier**: Percentage multiplier applied to total mult (`mult_multiplier`)
229///
230/// ## Resource Effects
231/// - **Money**: Coins awarded to the player (`money`)
232/// - **Hand Size**: Temporary hand size modification (`hand_size_mod`)
233/// - **Discards**: Temporary discard count modification (`discard_mod`)
234///
235/// ## Special Effects
236/// - **Retrigger**: Number of times to retrigger the effect (`retrigger`)
237/// - **Self Destruction**: Whether this joker destroys itself (`destroy_self`)
238/// - **Other Destruction**: Other jokers to destroy (`destroy_others`)
239/// - **Card Transformation**: Cards to transform into other cards (`transform_cards`)
240/// - **Messages**: Custom messages to display (`message`)
241///
242/// # Application Order
243///
244/// Effects are applied during scoring in this order:
245/// 1. **Chips**: Base chips + all chip bonuses
246/// 2. **Mult**: Base mult + all mult bonuses
247/// 3. **Mult Multipliers**: Apply all multipliers to total mult
248/// 4. **Final Score**: (Total Chips) × (Total Mult)
249/// 5. **Resources**: Award money and apply modifiers
250/// 6. **Special Effects**: Handle retriggering, destruction, transformation
251///
252/// # Examples
253///
254/// ```rust,ignore
255/// use balatro_rs::joker::JokerEffect;
256///
257/// // Simple scoring bonus
258/// let effect = JokerEffect::new().with_mult(5);
259///
260/// // Combined scoring effect
261/// let effect = JokerEffect::new()
262///     .with_chips(50)
263///     .with_mult(3)
264///     .with_money(2);
265///
266/// // Multiplicative effect
267/// let effect = JokerEffect::new().with_mult_multiplier(1.5); // +50% mult
268///
269/// // Complex effect with multiplicative bonus
270/// let effect = JokerEffect::new()
271///     .with_mult(10)
272///     .with_mult_multiplier(2.0);
273/// ```
274///
275/// # Performance Notes
276///
277/// `JokerEffect` instances are created frequently during scoring. The builder
278/// pattern methods consume and return `self` to enable efficient method chaining
279/// without additional allocations.
280#[derive(Debug, Clone, Serialize, Deserialize)]
281#[cfg_attr(feature = "python", pyclass)]
282pub struct JokerEffect {
283    /// Additional chips to add to the hand's base chips.
284    ///
285    /// Applied before mult calculations. Positive values increase the score.
286    pub chips: i32,
287
288    /// Additional mult to add to the hand's base mult.
289    ///
290    /// Applied before mult multipliers. Positive values increase the score.
291    pub mult: i32,
292
293    /// Money to award to the player.
294    ///
295    /// Applied immediately after hand scoring. Positive values give money.
296    pub money: i32,
297
298    /// Additional interest to add to the base interest calculation.
299    ///
300    /// This bonus is added to the base interest before the interest cap is applied.
301    /// Used by jokers like "To the Moon" that provide extra interest.
302    pub interest_bonus: i32,
303
304    /// Multiplier to apply to the total mult.
305    ///
306    /// Applied after all mult bonuses are summed. For example:
307    /// - 1.0 = no change (100%)
308    /// - 1.5 = +50% mult
309    /// - 2.0 = double mult (200%)
310    pub mult_multiplier: f64,
311
312    /// Number of times to retrigger this effect.
313    ///
314    /// A value of 2 means the effect triggers 3 times total (1 + 2 retriggers).
315    pub retrigger: u32,
316
317    /// Whether this joker should destroy itself after applying the effect.
318    ///
319    /// Used for one-time jokers or jokers with limited uses.
320    pub destroy_self: bool,
321
322    /// Other jokers to destroy when this effect is applied.
323    ///
324    /// Used for jokers that consume other jokers for powerful effects.
325    pub destroy_others: Vec<JokerId>,
326
327    /// Cards to transform into other cards.
328    ///
329    /// Each tuple represents (source_card, target_card). Used for jokers
330    /// that modify the deck or hand composition.
331    pub transform_cards: Vec<(Card, Card)>,
332
333    /// Temporary modification to hand size.
334    ///
335    /// Positive values increase hand size, negative values decrease it.
336    /// Applied for the current hand only.
337    pub hand_size_mod: i32,
338
339    /// Temporary modification to discard count.
340    ///
341    /// Positive values give extra discards, negative values reduce discards.
342    /// Applied for the current round only.
343    pub discard_mod: i32,
344
345    /// Sell value increase for this joker
346    pub sell_value_increase: i32,
347
348    /// Custom message to display to the player.
349    ///
350    /// Used for jokers with special effects, Easter eggs, or important notifications.
351    pub message: Option<String>,
352}

570/// Core trait that all jokers must implement.
571///
572/// This trait defines the interface for all joker implementations in the Balatro-RS system.
573/// Jokers can modify game scoring through lifecycle hooks and provide static information
574/// about their identity and behavior.
575///
576/// # Implementation Patterns
577///
578/// There are several patterns for implementing jokers:
579///
580/// ## 1. Direct Implementation
581/// For complex jokers requiring custom logic:
582/// ```rust,ignore
583/// use balatro_rs::joker::{Joker, JokerId, JokerRarity, JokerEffect, GameContext};
584/// use balatro_rs::card::Card;
585/// use balatro_rs::hand::SelectHand;
586///
587/// #[derive(Debug)]
588/// struct CustomJoker;
589///
590/// impl Joker for CustomJoker {
591///     fn id(&self) -> JokerId { JokerId::CustomJoker }
592///     fn name(&self) -> &str { "Custom Joker" }
593///     fn description(&self) -> &str { "Complex custom logic" }
594///     fn rarity(&self) -> JokerRarity { JokerRarity::Common }
595///
596///     fn on_card_scored(&self, context: &mut GameContext, card: &Card) -> JokerEffect {
597///         // Custom scoring logic
598///         if self.complex_condition(context, card) {
599///             JokerEffect::new().with_mult(5)
600///         } else {
601///             JokerEffect::new()
602///         }
603///     }
604/// }
605/// ```
606///
607/// ## 2. Static Joker Framework
608/// For simple conditional jokers:
609/// ```rust,ignore
610/// use balatro_rs::static_joker::{StaticJoker, StaticCondition};
611/// use balatro_rs::card::Suit;
612///
613/// let greedy_joker = StaticJoker::builder(
614///     JokerId::GreedyJoker,
615///     "Greedy Joker",
616///     "+3 Mult per Diamond"
617/// )
618/// .rarity(JokerRarity::Common)
619/// .mult(3)
620/// .condition(StaticCondition::SuitScored(Suit::Diamond))
621/// .per_card()
622/// .build()?;
623/// ```
624///
625/// # Lifecycle Events
626///
627/// Jokers integrate with the game through well-defined lifecycle events that mirror
628/// Balatro's game flow:
629///
630/// 1. **Blind Start**: `on_blind_start()` - Called when a new blind begins
631/// 2. **Hand Play**: `on_hand_played()` - Called when a hand is played for scoring
632/// 3. **Card Scoring**: `on_card_scored()` - Called for each individual scoring card
633/// 4. **Discard**: `on_discard()` - Called when cards are discarded
634/// 5. **Shop**: `on_shop_open()` - Called when entering the shop
635/// 6. **Round End**: `on_round_end()` - Called at the end of each round
636///
637/// # Effect Application
638///
639/// Joker effects are applied in a specific order to ensure consistent behavior:
640/// 1. Chips calculation: Base chips + chip bonuses
641/// 2. Mult calculation: Base mult + mult bonuses
642/// 3. Mult multiplication: Apply mult multipliers
643/// 4. Final score: chips × mult
644///
645/// # Thread Safety
646///
647/// All joker implementations must be `Send + Sync` to support concurrent access
648/// in multi-threaded environments and RL training scenarios.
649///
650/// # Performance Considerations
651///
652/// - `on_card_scored()` is called most frequently and should be optimized
653/// - Use early returns for non-matching conditions
654/// - Avoid expensive operations in hot paths
655/// - Consider caching expensive calculations in joker state
656pub trait Joker: Send + Sync + std::fmt::Debug {
657    /// Get the unique identifier for this joker.
658    ///
659    /// This ID must be unique across all joker implementations and is used for
660    /// state management, serialization, and factory creation.
661    ///
662    /// # Returns
663    /// The unique `JokerId` for this joker type.
664    fn id(&self) -> JokerId;
665
666    /// Get the display name of this joker.
667    ///
668    /// This should be a human-readable name that matches the Balatro naming convention.
669    ///
670    /// # Returns
671    /// A string slice containing the joker's name.
672    fn name(&self) -> &str;
673
674    /// Get the description of this joker's effect.
675    ///
676    /// This should clearly describe what the joker does in a concise format
677    /// suitable for display in the user interface.
678    ///
679    /// # Returns
680    /// A string slice containing the joker's description.
681    fn description(&self) -> &str;
682
683    /// Get the rarity level of this joker.
684    ///
685    /// Rarity affects the base cost and availability in shops.
686    ///
687    /// # Returns
688    /// The `JokerRarity` level for this joker.
689    fn rarity(&self) -> JokerRarity;
690
691    /// Get the base cost of this joker in the shop.
692    ///
693    /// The default implementation uses rarity-based pricing:
694    /// - Common: 3 coins
695    /// - Uncommon: 6 coins
696    /// - Rare: 8 coins
697    /// - Legendary: 20 coins
698    ///
699    /// Override this method for jokers with custom pricing.
700    ///
701    /// # Returns
702    /// The cost in coins to purchase this joker.
703    fn cost(&self) -> usize {
704        match self.rarity() {
705            JokerRarity::Common => 3,
706            JokerRarity::Uncommon => 6,
707            JokerRarity::Rare => 8,
708            JokerRarity::Legendary => 20,
709        }
710    }
711
712    /// Get the current sell value (defaults to cost / 2)
713    fn sell_value(&self, accumulated_bonus: f64) -> usize {
714        (self.cost() / 2) + (accumulated_bonus as usize)
715    }
716
717    // Lifecycle hooks with default implementations
718
719    /// Called when a hand is played and scored.
720    ///
721    /// This is the primary hook for jokers that provide per-hand effects,
722    /// such as bonuses for specific hand types (Pair, Flush, etc.).
723    ///
724    /// # Arguments
725    /// * `context` - Mutable reference to the current game context
726    /// * `hand` - The hand being played and scored
727    ///
728    /// # Returns
729    /// A `JokerEffect` describing any bonuses to apply to the hand.
730    ///
731    /// # Example
732    /// ```rust,ignore
733    /// fn on_hand_played(&self, context: &mut GameContext, hand: &SelectHand) -> JokerEffect {
734    ///     if hand.is_pair().is_some() {
735    ///         JokerEffect::new().with_mult(8)  // +8 mult for pairs
736    ///     } else {
737    ///         JokerEffect::new()
738    ///     }
739    /// }
740    /// ```
741    fn on_hand_played(&self, _context: &mut GameContext, _hand: &SelectHand) -> JokerEffect {
742        JokerEffect::new()
743    }
744
745    /// Called for each card as it's scored.
746    ///
747    /// This is the primary hook for jokers that provide per-card effects,
748    /// such as bonuses for specific suits, ranks, or card properties.
749    ///
750    /// # Arguments
751    /// * `context` - Mutable reference to the current game context
752    /// * `card` - The individual card being scored
753    ///
754    /// # Returns
755    /// A `JokerEffect` describing any bonuses to apply for this card.
756    ///
757    /// # Performance Note
758    /// This method is called frequently (once per scoring card) and should be optimized.
759    /// Use early returns for non-matching conditions.
760    ///
761    /// # Example
762    /// ```rust,ignore
763    /// fn on_card_scored(&self, context: &mut GameContext, card: &Card) -> JokerEffect {
764    ///     if card.suit == Suit::Diamond {
765    ///         JokerEffect::new().with_mult(3)  // +3 mult per Diamond
766    ///     } else {
767    ///         JokerEffect::new()
768    ///     }
769    /// }
770    /// ```
771    fn on_card_scored(&self, _context: &mut GameContext, _card: &Card) -> JokerEffect {
772        JokerEffect::new()
773    }
774
775    /// Called when a new blind starts.
